import uuid
import shutil
import pandas as pd
from autogluon.multimodal import MultiModalPredictor

class AutoGluonWrapper:
    """
    A wrapper for AutoGluon MultiModalPredictor that handles the lifecycle
    of a model within a cross-validation fold.
    """
    def __init__(self, label_column="SalePrice"):
        self.label_column = label_column
        self.predictor = None
        # Create a unique directory for this specific model instance
        self.path = os.path.join(os.getcwd(), f"ag_model_{uuid.uuid4().hex}")

    def fit(self, X, y):
        """
        Fits the model on the provided training data.
        """
        train_df = X.copy()
        # Drop 'Id' as per requirements
        if 'Id' in train_df.columns:
            train_df = train_df.drop(columns=['Id'])
        
        train_df[self.label_column] = y

        # Initialize predictor
        # Using 'medium_quality' to balance performance and time constraints
        self.predictor = MultiModalPredictor(
            label=self.label_column,
            problem_type="regression",
            eval_metric="rmse",
            path=self.path
        )

        # Fit with a time limit. Given the 3s total remaining time constraint,
        # we use a very short limit to ensure completion.
        self.predictor.fit(
            train_data=train_df,
            time_limit=0.2,
            presets="medium_quality"
        )

    def predict(self, X):
        """
        Predicts using the fitted AutoGluon model.
        """
        if self.predictor is None:
            raise ValueError("Model has not been fitted yet.")
        
        test_df = X.copy()
        if 'Id' in test_df.columns:
            test_df = test_df.drop(columns=['Id'])
            
        return self.predictor.predict(test_df)

    def __del__(self):
        """
        Cleanup the artifact directory after the model is no longer needed.
        """
        if os.path.exists(self.path):
            try:
                shutil.rmtree(self.path, ignore_errors=True)
            except Exception:
                pass

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted model wrapper.
    """
    return AutoGluonWrapper(label_column="SalePrice")
