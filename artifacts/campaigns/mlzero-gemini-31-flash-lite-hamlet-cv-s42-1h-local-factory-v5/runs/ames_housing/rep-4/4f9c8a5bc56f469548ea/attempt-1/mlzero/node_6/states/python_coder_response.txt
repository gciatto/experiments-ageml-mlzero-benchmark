```python
import os
import uuid
import shutil
import pandas as pd
from autogluon.tabular import TabularPredictor

class AutoGluonRegressorWrapper:
    """
    A wrapper for AutoGluon TabularPredictor designed for cross-validation.
    Enforces sequential execution and resource constraints to prevent
    segmentation faults and resource contention in restricted environments.
    """
    def __init__(self, label='SalePrice'):
        self.label = label
        self.predictor = None
        # Unique directory per fold to prevent file locking/conflicts
        self.path = os.path.join(os.getcwd(), f"ag_model_{uuid.uuid4().hex}")

    def fit(self, X, y):
        """
        Fits the AutoGluon predictor using sequential execution.
        """
        train_data = X.copy()
        train_data[self.label] = y

        # Disable multi-processing and Ray to avoid segmentation faults.
        # Use 'best_quality' for a balance of performance and stability.
        # Time limit is set conservatively to ensure all 10 folds complete.
        self.predictor = TabularPredictor(
            label=self.label,
            path=self.path,
            eval_metric='rmse'
        ).fit(
            train_data,
            presets='best_quality',
            time_limit=2,  # Strict limit to ensure completion within 26s total
            num_cpus=1,
            num_gpus=0,
            verbosity=0,
            ag_args_fit={'num_cpus': 1}
        )
        return self

    def predict(self, X):
        """
        Predicts using the fitted AutoGluon predictor.
        """
        if self.predictor is None:
            raise ValueError("Model has not been fitted yet.")
        return self.predictor.predict(X)

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted AutoGluonRegressorWrapper.
    """
    return AutoGluonRegressorWrapper(label='SalePrice')
```
