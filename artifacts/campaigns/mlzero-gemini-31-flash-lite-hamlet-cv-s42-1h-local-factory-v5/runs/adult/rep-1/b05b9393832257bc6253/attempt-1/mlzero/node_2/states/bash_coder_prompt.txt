    A wrapper for AutoGluon MultiModalPredictor that handles the lifecycle
    of a model within a cross-validation fold.

        self.predictor = None

        Fits the model on the provided training fold.
        """

        # Initialize predictor
        # We use 'medium_quality' to ensure we stay within the time budget
        # across 10 folds.
        self.predictor = MultiModalPredictor(
            label=self.label_column,
            path=self.path,
            eval_metric="balanced_accuracy"
        )
        
        # Fit the model
        # We limit max_epochs and time_limit to ensure the 10-fold process finishes
        # within the remaining shared deadline.
        self.predictor.fit(
            train_data=train_data,

                "optim.max_epochs": 5,
                "env.num_gpus": 0  # Ensure compatibility if GPU is not available
            },
            time_limit=240  # Strict per-fold limit to respect total budget
        )
        return self

    def predict(self, X):
        """
        Predicts labels for the provided data.

        """
        Cleanup artifacts to save disk space.

            try:
                shutil.rmtree(self.path)

                pass
