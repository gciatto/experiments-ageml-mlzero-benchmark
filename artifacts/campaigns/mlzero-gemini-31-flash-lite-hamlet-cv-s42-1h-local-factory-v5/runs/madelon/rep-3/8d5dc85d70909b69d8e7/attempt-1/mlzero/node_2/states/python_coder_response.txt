```python
import os
import uuid
import shutil
import pandas as pd
from autogluon.multimodal import MultiModalPredictor

class AutoGluonWrapper:
    """
    A wrapper for AutoGluon MultiModalPredictor that handles the lifecycle
    of the model within the constraints of the cross-validation harness.
    """
    def __init__(self, label_col='Class'):
        self.label_col = label_col
        self.predictor = None
        self.model_path = os.path.join(os.getcwd(), f"autogluon_model_{uuid.uuid4().hex}")

    def fit(self, X, y):
        """
        Fits the MultiModalPredictor on the provided training fold.
        """
        # Combine features and target for AutoGluon
        train_df = X.copy()
        train_df[self.label_col] = y

        # Initialize predictor
        # Using 'medium_quality' to ensure we stay within the total time budget
        # for all 10 folds.
        self.predictor = MultiModalPredictor(
            label=self.label_col,
            path=self.model_path,
            eval_metric='balanced_accuracy'
        )

        # Fit with a time limit that allows for 10 folds to complete
        # Total budget is ~3259s, so we allocate ~280s per fold to be safe.
        self.predictor.fit(
            train_data=train_df,
            time_limit=280,
            presets='medium_quality'
        )

    def predict(self, X):
        """
        Predicts labels for the provided validation fold.
        """
        if self.predictor is None:
            raise ValueError("Model has not been fitted yet.")
        
        return self.predictor.predict(X)

    def __del__(self):
        """
        Cleanup the artifact directory after the object is destroyed.
        """
        if self.model_path and os.path.exists(self.model_path):
            try:
                shutil.rmtree(self.model_path)
            except Exception:
                pass

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted AutoGluonWrapper.
    """
    # The label column is 'Class' as per the task description
    return AutoGluonWrapper(label_col='Class')
```
