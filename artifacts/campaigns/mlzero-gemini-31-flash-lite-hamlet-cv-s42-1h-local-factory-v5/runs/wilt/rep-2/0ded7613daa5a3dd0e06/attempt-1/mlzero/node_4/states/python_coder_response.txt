import numpy as np
from sklearn.ensemble import HistGradientBoostingClassifier
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.base import BaseEstimator, ClassifierMixin

class WiltClassifierWrapper(BaseEstimator, ClassifierMixin):
    """
    A wrapper for a HistGradientBoostingClassifier.
    This model is chosen for its speed and efficiency in handling tabular data
    within strict time constraints, while providing robust performance.
    """
    def __init__(self, random_state=42):
        self.random_state = random_state
        # HistGradientBoostingClassifier is natively fast and handles 
        # missing values and scaling internally, but we include a 
        # StandardScaler for consistency with standard ML pipelines.
        self.model = Pipeline([
            ('scaler', StandardScaler()),
            ('clf', HistGradientBoostingClassifier(
                max_iter=100,
                learning_rate=0.1,
                max_depth=8,
                random_state=self.random_state,
                class_weight='balanced',
                early_stopping=True
            ))
        ])

    def fit(self, X, y):
        """
        Fits the pipeline on the provided training data.
        """
        self.model.fit(X, y)
        return self

    def predict(self, X):
        """
        Predicts class labels for the provided data.
        """
        return self.model.predict(X)

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted instance of the classifier.
    
    Parameters:
    X_train: Training features (provided for context)
    y_train: Training labels (provided for context)
    
    Returns:
    An unfitted WiltClassifierWrapper instance.
    """
    # We return a fresh instance for every fold to ensure no state leakage.
    return WiltClassifierWrapper(random_state=42)
