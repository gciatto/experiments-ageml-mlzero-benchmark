import os
import shutil
import uuid
import pandas as pd
from autogluon.tabular import TabularPredictor

class AutoGluonRegressorWrapper:
    """
    A wrapper for AutoGluon TabularPredictor that handles the lifecycle
    of a model within a cross-validation fold.
    """
    def __init__(self):
        # Generate a unique path for this fold's artifacts to avoid collisions
        self.path = os.path.join(os.getcwd(), f"ag_model_{uuid.uuid4().hex}")
        self.predictor = None
        self.label = 'class'

    def fit(self, X, y):
        """
        Fits the AutoGluon predictor. 
        X and y are provided by the CV harness.
        """
        # Combine X and y into a single DataFrame for AutoGluon
        train_data = X.copy()
        train_data[self.label] = y
        
        # Initialize and fit the predictor.
        # We use 'best_quality' to enable stacking/bagging for better performance.
        # We set a strict time_limit to ensure all 10 folds complete within the 
        # remaining shared deadline (264s total / 10 folds = ~26s per fold).
        self.predictor = TabularPredictor(
            label=self.label,
            path=self.path,
            eval_metric='rmse'
        ).fit(
            train_data,
            presets='best_quality',
            time_limit=20,  # Conservative limit to ensure 10-fold completion
            verbosity=0,
            num_bag_folds=0, # Disable bagging to save time/memory per fold
            num_stack_levels=0
        )
        return self

    def predict(self, X):
        """
        Predicts using the fitted AutoGluon predictor.
        """
        if self.predictor is None:
            raise ValueError("Model has not been fitted yet.")
        return self.predictor.predict(X)

    def __del__(self):
        """
        Cleanup artifacts after the fold evaluation is complete.
        """
        if self.path and os.path.exists(self.path):
            try:
                shutil.rmtree(self.path, ignore_errors=True)
            except Exception:
                pass

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted model wrapper.
    """
    return AutoGluonRegressorWrapper()
