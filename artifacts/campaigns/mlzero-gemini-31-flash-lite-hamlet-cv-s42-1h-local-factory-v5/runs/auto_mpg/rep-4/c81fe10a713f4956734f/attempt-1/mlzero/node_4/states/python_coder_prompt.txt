import os
import shutil
import tempfile
import numpy as np
import pandas as pd

import torch
from autogluon.tabular import TabularPredictor
from sklearn.base import BaseEstimator, RegressorMixin

class AutoGluonRegressorWrapper(BaseEstimator, RegressorMixin):
    """
    A wrapper for AutoGluon TabularPredictor that adheres to the 
    scikit-learn estimator interface for use in cross-validation.
    """
    def __init__(self, seed=42):
        self.seed = seed
        self.predictor = None
        self.temp_dir = None
        self.target_col = "class"

    def _set_seeds(self):
        random.seed(self.seed)
        np.random.seed(self.seed)
        torch.manual_seed(self.seed)

    def fit(self, X, y):
        # Create a fresh directory for this fold's artifacts
        self.temp_dir = tempfile.mkdtemp(prefix="ag_fold_")
        
        # Prepare data
        df = X.copy()
        df[self.target_col] = y
        
        self._set_seeds()
        
        # Initialize and fit predictor
        # Using 'best_quality' but constrained by time_limit to ensure 
        # all 10 folds complete within the shared deadline.
        self.predictor = TabularPredictor(
            label=self.target_col,
            path=self.temp_dir,
            problem_type='regression',
            verbosity=0
        )
        
        self.predictor.fit(
            df,
            time_limit=240,  # Conservative limit to ensure 10 folds fit in ~3500s
            presets='medium_quality',
            ag_args_fit={'random_seed': self.seed}
        )
        return self

    def predict(self, X):
        if self.predictor is None:
            raise ValueError("Model must be fitted before calling predict.")
        return self.predictor.predict(X)

    def __del__(self):
        # Cleanup artifacts after use
        if self.temp_dir and os.path.exists(self.temp_dir):
            try:
                shutil.rmtree(self.temp_dir)
            except:
                pass

def build_model(X_train, y_train):

### Previous errors
ERROR_SUMMARY: The `ValueError` is caused by passing an unsupported keyword argument `random_seed` to the `TabularPredictor.fit()` method. AutoGluon's `fit` method does not accept `random_seed` as a direct argument; instead, stochastic behavior is controlled via the `ag_args_fit` dictionary or by setting the seed globally before calling `fit`.

SUGGESTED_FIX: Remove `random_seed=self.seed` from the `fit()` call in your `AutoGluonRegressorWrapper`. To ensure reproducibility, set the seed globally using `import random; random.seed(self.seed)`, `import numpy as np; np.random.seed(self.seed)`, and `import torch; torch.manual_seed(self.seed)` immediately before calling `fit()`.
