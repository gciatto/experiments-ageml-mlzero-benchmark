import numpy as np
import pandas as pd
from sklearn.base import BaseEstimator, RegressorMixin
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.ensemble import HistGradientBoostingRegressor

class AutoMPGRegressor(BaseEstimator, RegressorMixin):
    """
    A robust regression wrapper for the auto_mpg dataset.
    Uses HistGradientBoostingRegressor which is highly efficient for tabular data.
    Includes automated feature preprocessing for numerical and categorical inputs.
    """
    def __init__(self, random_state=42):
        self.random_state = random_state
        self.model = None

    def _build_pipeline(self, X):
        # Identify numerical and categorical columns
        # 'origin' is categorical, others are numerical
        numeric_features = [col for col in X.columns if col != 'origin']
        categorical_features = ['origin'] if 'origin' in X.columns else []

        preprocessor = ColumnTransformer(
            transformers=[
                ('num', StandardScaler(), numeric_features),
                ('cat', OneHotEncoder(handle_unknown='ignore'), categorical_features)
            ])

        # HistGradientBoostingRegressor is fast and handles large datasets well
        # We limit max_iter to ensure we stay within the strict time budget
        return Pipeline([
            ('preprocessor', preprocessor),
            ('regressor', HistGradientBoostingRegressor(
                random_state=self.random_state,
                max_iter=100,
                early_stopping=True,
                validation_fraction=0.1,
                n_iter_no_change=10
            ))
        ])

    def fit(self, X, y):
        """
        Fits the pipeline on the provided training data.
        """
        self.model = self._build_pipeline(X)
        self.model.fit(X, y)
        return self

    def predict(self, X):
        """
        Predicts using the fitted pipeline.
        """
        if self.model is None:
            raise ValueError("Model must be fitted before calling predict.")
        return self.model.predict(X)
