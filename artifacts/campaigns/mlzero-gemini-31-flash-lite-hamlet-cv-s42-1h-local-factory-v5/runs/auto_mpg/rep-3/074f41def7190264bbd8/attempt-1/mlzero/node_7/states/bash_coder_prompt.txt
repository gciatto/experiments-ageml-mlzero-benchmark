                random_state=self.random_state,
                max_iter=100,
                learning_rate=0.05,
                early_stopping=True,
                validation_fraction=0.1,
                n_iter_no_change=10
            ))
        ])

        """
        Fits the pipeline on the provided training data.

                                         'weight', 'acceleration', 'model', 'origin'])
        self.pipeline.fit(X, y)
        return self

    def predict(self, X):
        """
        Predicts using the fitted pipeline.

                                         'weight', 'acceleration', 'model', 'origin'])
