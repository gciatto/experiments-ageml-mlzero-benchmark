import numpy as np
import pandas as pd
from sklearn.base import BaseEstimator, RegressorMixin
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.impute import SimpleImputer
from sklearn.ensemble import HistGradientBoostingRegressor

class AutoMPGRegressor(BaseEstimator, RegressorMixin):
    """
    A robust regression wrapper for the auto_mpg dataset.
    Uses HistGradientBoostingRegressor with automated preprocessing.
    """
    def __init__(self, random_state=42):
        self.random_state = random_state
        self.model = None

    def _build_pipeline(self):
        # Identify numeric and categorical features
        # Based on dataset: cylinders, displacement, horsepower, weight, acceleration, model, origin
        numeric_features = ['cylinders', 'displacement', 'horsepower', 'weight', 'acceleration', 'model']
        categorical_features = ['origin']

        numeric_transformer = Pipeline(steps=[
            ('imputer', SimpleImputer(strategy='median')),
            ('scaler', StandardScaler())
        ])

        categorical_transformer = Pipeline(steps=[
            ('imputer', SimpleImputer(strategy='most_frequent')),
            ('onehot', OneHotEncoder(handle_unknown='ignore'))
        ])

        preprocessor = ColumnTransformer(
            transformers=[
                ('num', numeric_transformer, numeric_features),
                ('cat', categorical_transformer, categorical_features)
            ])

        return Pipeline(steps=[
            ('preprocessor', preprocessor),
            ('regressor', HistGradientBoostingRegressor(
                random_state=self.random_state,
                max_iter=200,
                learning_rate=0.05,
                max_leaf_nodes=31,
                early_stopping=True,
                validation_fraction=0.1,
                n_iter_no_change=10
            ))
        ])

    def fit(self, X, y):
        """
        Fits the pipeline on the provided training data.
        """
        self.model = self._build_pipeline()
        self.model.fit(X, y)
        return self

    def predict(self, X):
        """
        Predicts using the fitted pipeline.
        """
        if self.model is None:
            raise ValueError("Model must be fitted before calling predict.")
        return self.model.predict(X)
