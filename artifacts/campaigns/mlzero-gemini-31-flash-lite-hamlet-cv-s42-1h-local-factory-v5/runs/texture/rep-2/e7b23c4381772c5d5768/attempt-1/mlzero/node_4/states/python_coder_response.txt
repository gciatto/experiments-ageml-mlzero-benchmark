```python
import numpy as np
from sklearn.base import BaseEstimator, ClassifierMixin
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler, RobustScaler
from sklearn.ensemble import HistGradientBoostingClassifier
from sklearn.decomposition import PCA

class ModelWrapper(BaseEstimator, ClassifierMixin):
    """
    A robust classification wrapper using HistGradientBoostingClassifier.
    
    HistGradientBoostingClassifier is chosen for its speed and efficiency 
    with large datasets, and its native support for missing values and 
    internal parallelization control.
    """
    def __init__(self, random_state=42):
        self.random_state = random_state
        # We use a pipeline with robust scaling to handle potential outliers
        # and a PCA step to reduce dimensionality if features are highly correlated.
        # HistGradientBoostingClassifier is set to use 1 thread to avoid 
        # resource contention in the CV environment.
        self.pipeline = Pipeline([
            ('scaler', RobustScaler()),
            ('pca', PCA(n_components=0.95, random_state=self.random_state)),
            ('clf', HistGradientBoostingClassifier(
                max_iter=200,
                learning_rate=0.1,
                max_depth=10,
                random_state=self.random_state,
                early_stopping=True,
                scoring='balanced_accuracy'
            ))
        ])

    def fit(self, X, y):
        """
        Fits the pipeline on the provided training data.
        """
        # Ensure X is treated as a numpy array for consistency
        self.pipeline.fit(X, y)
        return self

    def predict(self, X):
        """
        Predicts class labels for the provided input.
        """
        return self.pipeline.predict(X)

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted model instance.
    
    Args:
        X_train: Training features (contextual).
        y_train: Training labels (contextual).
        
    Returns:
        An unfitted instance of ModelWrapper.
    """
    return ModelWrapper(random_state=42)
```
