```python
import os
import shutil
import tempfile
import pandas as pd
from autogluon.tabular import TabularPredictor

class AutoGluonWrapper:
    """
    A wrapper for AutoGluon TabularPredictor that handles the lifecycle
    of a model within a cross-validation fold.
    """
    def __init__(self):
        self.predictor = None
        # Create a unique directory for this fold's artifacts
        self.temp_dir = tempfile.mkdtemp(prefix="ag_fold_")
        self.label = "Class"

    def fit(self, X, y):
        # Combine X and y into a single DataFrame for AutoGluon
        train_data = X.copy()
        train_data[self.label] = y
        
        # We use a time-constrained fit. Given the strict 154s total budget 
        # for 10 folds, we allocate a very tight budget per fold.
        # 'best_quality' is used for performance, but time_limit is strictly 
        # capped to ensure the entire CV process completes.
        self.predictor = TabularPredictor(
            label=self.label,
            path=self.temp_dir,
            eval_metric='balanced_accuracy'
        ).fit(
            train_data,
            presets='best_quality',
            time_limit=12,  # Aggressive limit to fit within 154s total
            verbosity=0
        )
        return self

    def predict(self, X):
        if self.predictor is None:
            raise ValueError("Model has not been fitted yet.")
        return self.predictor.predict(X)

    def __del__(self):
        # Clean up the temporary directory after the fold is processed
        if self.temp_dir and os.path.exists(self.temp_dir):
            try:
                shutil.rmtree(self.temp_dir, ignore_errors=True)
            except Exception:
                pass

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted AutoGluonWrapper.
    """
    return AutoGluonWrapper()
```
