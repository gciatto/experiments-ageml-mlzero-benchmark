import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.base import BaseEstimator, ClassifierMixin

class ModelWrapper(BaseEstimator, ClassifierMixin):
    """
    A wrapper class that encapsulates preprocessing and a classifier.
    Uses a Random Forest with balanced class weights to handle potential
    imbalance and ensure robust performance on the texture dataset.
    """
    def __init__(self, random_state=42):
        self.random_state = random_state
        self.pipeline = Pipeline([
            ('scaler', StandardScaler()),
            ('clf', RandomForestClassifier(
                n_estimators=100,
                max_depth=15,
                class_weight='balanced',
                random_state=self.random_state,
                n_jobs=1  # Constrained to ensure stability within shared time
            ))
        ])

    def fit(self, X, y):
        """
        Fits the pipeline on the provided training data.
        """
        self.pipeline.fit(X, y)
        return self

    def predict(self, X):
        """
        Predicts class labels using the fitted pipeline.
        """
        return self.pipeline.predict(X)

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted model instance.
    
    Parameters:
    X_train: Training features (provided for context, though not used for fitting here)
    y_train: Training labels (provided for context)
    
    Returns:
    An unfitted instance of ModelWrapper.
    """
    return ModelWrapper(random_state=42)
