import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.feature_selection import VarianceThreshold

class HARModelWrapper:
    """
    A wrapper for a classification model that handles preprocessing 
    and prediction for the HAR dataset.
    """
    def __init__(self):
        # Using a robust ensemble method suitable for high-dimensional sensor data.
        # VarianceThreshold removes constant features, StandardScaler normalizes,
        # and RandomForest provides strong baseline performance.
        self.model = Pipeline([
            ('selector', VarianceThreshold(threshold=0.01)),
            ('scaler', StandardScaler()),
            ('classifier', RandomForestClassifier(
                n_estimators=100,
                max_depth=20,
                random_state=42,
                n_jobs=-1,
                class_weight='balanced'
            ))
        ])

    def fit(self, X, y):
        """
        Fits the pipeline on the provided training data.
        """
        self.model.fit(X, y)
        return self

    def predict(self, X):
        """
        Predicts class labels for the provided input data.
        """
        return self.model.predict(X)

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted model instance.
    
    Args:
        X_train: Training features (not used for initialization, 
                 but required by contract).
        y_train: Training labels (not used for initialization, 
                 but required by contract).
                 
    Returns:
        An instance of HARModelWrapper.
    """
    return HARModelWrapper()
