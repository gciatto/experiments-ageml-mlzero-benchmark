```python
import os
import shutil
import uuid
import pandas as pd
from autogluon.multimodal import MultiModalPredictor

class AutoGluonWrapper:
    """
    A wrapper for AutoGluon MultiModalPredictor that handles the lifecycle
    of the predictor for each fold, ensuring clean artifact directories.
    """
    def __init__(self, label_column='Class'):
        self.label_column = label_column
        self.predictor = None
        self.path = f"autogluon_model_{uuid.uuid4().hex}"

    def fit(self, X, y):
        # Combine X and y into a single DataFrame for AutoGluon
        train_data = X.copy()
        train_data[self.label_column] = y
        
        # Initialize predictor
        # Using 'medium_quality' to ensure we stay within the total time budget
        # for 10 folds.
        self.predictor = MultiModalPredictor(
            label=self.label_column,
            path=self.path,
            eval_metric='balanced_accuracy'
        )
        
        # Fit the model
        # We limit time_limit to ensure all 10 folds finish within the 1489s budget.
        # 1489s / 10 folds = ~148s per fold. We set to 120s to allow for overhead.
        self.predictor.fit(
            train_data=train_data,
            time_limit=120,
            presets='medium_quality'
        )
        return self

    def predict(self, X):
        if self.predictor is None:
            raise ValueError("Model has not been fitted yet.")
        return self.predictor.predict(X)

    def __del__(self):
        # Clean up artifact directory to save disk space
        if os.path.exists(self.path):
            try:
                shutil.rmtree(self.path)
            except Exception:
                pass

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted model wrapper.
    """
    # The target column is 'Class' as per the task description.
    return AutoGluonWrapper(label_column='Class')
```
