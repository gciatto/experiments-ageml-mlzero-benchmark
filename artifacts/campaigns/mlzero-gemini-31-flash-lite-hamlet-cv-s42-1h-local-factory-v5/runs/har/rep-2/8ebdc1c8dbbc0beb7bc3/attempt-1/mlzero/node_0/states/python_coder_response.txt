```python
import os
import shutil
import tempfile
import uuid
import pandas as pd
from autogluon.tabular import TabularPredictor

class AutoGluonWrapper:
    """
    A wrapper for AutoGluon TabularPredictor that handles the lifecycle
    of a model within a cross-validation fold.
    """
    def __init__(self):
        self.predictor = None
        self.temp_dir = tempfile.mkdtemp(prefix="ag_fold_")
        self.label = 'Class'

    def fit(self, X, y):
        """
        Fits the AutoGluon predictor on the provided training fold.
        """
        # Combine X and y for AutoGluon
        train_data = X.copy()
        train_data[self.label] = y
        
        # We use a fresh directory for each fold to ensure isolation
        # and clean up previous artifacts if any.
        fold_path = os.path.join(self.temp_dir, str(uuid.uuid4()))
        
        # Initialize and fit the predictor
        # Using 'extreme' preset as requested.
        # We limit time to ensure all 10 folds finish within the total budget.
        # Given 3570s total, we allocate ~300s per fold.
        self.predictor = TabularPredictor(
            label=self.label,
            path=fold_path,
            eval_metric='balanced_accuracy'
        ).fit(
            train_data,
            presets='extreme',
            time_limit=300,
            verbosity=0
        )

    def predict(self, X):
        """
        Predicts class labels for the provided data.
        """
        if self.predictor is None:
            raise ValueError("Model has not been fitted yet.")
        return self.predictor.predict(X)

    def __del__(self):
        """
        Cleanup temporary directory.
        """
        if os.path.exists(self.temp_dir):
            shutil.rmtree(self.temp_dir, ignore_errors=True)

def build_model(X_train, y_train):
    """
    Factory function to return a fresh, unfitted AutoGluonWrapper.
    """
    return AutoGluonWrapper()
```
