```bash
#!/bin/bash
set -euo pipefail

# Define environment path and requirements file paths
ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/california_housing/rep-2/353a64eebb2bcfbd2485/attempt-1/mlzero/node_5/conda_env"
REQ_FILE_TABULAR="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"
REQ_FILE_COMMON="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"

echo "--- Starting Conda environment setup ---"

# Create the Conda environment with Python 3.11
echo "Creating conda environment at $ENV_PATH with Python 3.11..."
conda create -p "$ENV_PATH" python=3.11 --yes || { echo "Error: Failed to create conda environment."; exit 1; }

# Activate the environment
echo "Activating conda environment..."
# Initialize conda for the current shell session if not already done
source "$(conda info --base)/etc/profile.d/conda.sh" || { echo "Error: conda.sh not found or failed to source."; exit 1; }
conda activate "$ENV_PATH" || { echo "Error: Failed to activate conda environment $ENV_PATH."; exit 1; }

# Install uv
echo "Installing uv into the environment..."
pip install uv || { echo "Error: Failed to install uv."; exit 1; }

# Install required packages from requirements files using uv
echo "Installing required packages from requirements files using uv..."
# The command explicitly includes --prerelease=allow and the specified order of requirements files.
# uv respects versions in requirements.txt and avoids unnecessary upgrades/reinstalls.
uv pip install -r "$REQ_FILE_TABULAR" --prerelease=allow -r "$REQ_FILE_COMMON" || { echo "Error: Failed to install required packages."; exit 1; }

echo "--- Conda environment setup complete ---"
```
