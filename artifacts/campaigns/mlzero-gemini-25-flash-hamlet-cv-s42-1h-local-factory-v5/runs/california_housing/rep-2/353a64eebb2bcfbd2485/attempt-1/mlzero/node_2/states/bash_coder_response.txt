```bash
#!/bin/bash

# Exit immediately if a command exits with a non-zero status.
set -e

# Define the target conda environment path
ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/california_housing/rep-2/353a64eebb2bcfbd2485/attempt-1/mlzero/node_2/conda_env"

# Define paths to the requirements files
COMMON_REQ_PATH="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
MULTIMODAL_REQ_PATH="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.multimodal/requirements.txt"

echo "Starting minimal BASH INSTALLATION for AutoGluon MultiModal..."

# Ensure conda is initialized in the current shell session
# This is crucial for scripts that might not inherit a fully initialized conda environment.
if ! type "conda" > /dev/null; then
    echo "Conda command not found. Please ensure Anaconda/Miniconda is installed and initialized."
    exit 1
fi
source "$(conda info --base)/etc/profile.d/conda.sh"

# Create the conda environment with Python 3.11
echo "Creating conda environment at '$ENV_PATH' with Python 3.11..."
conda create --prefix "$ENV_PATH" python=3.11 -y

# Activate the newly created environment
echo "Activating conda environment '$ENV_PATH'..."
conda activate "$ENV_PATH"

# Install uv (a fast Python package installer)
echo "Installing uv..."
pip install uv

# Install required packages from the specified requirements files using uv
# --prerelease=allow is used as AutoGluon often utilizes pre-release versions.
echo "Installing core and AutoGluon MultiModal requirements using uv..."
uv pip install -r "$MULTIMODAL_REQ_PATH" --prerelease=allow -r "$COMMON_REQ_PATH"

# Install numpy, pandas, and scikit-learn explicitly as required
echo "Installing numpy, pandas, and scikit-learn using uv..."
uv pip install numpy pandas scikit-learn

echo "Conda environment setup complete at '$ENV_PATH'."
echo "The environment is now ready for the factory module."
```
