```python
import pandas as pd
import numpy as np
import os
import uuid
from pathlib import Path
from autogluon.multimodal import MultiModalPredictor

# Module-level constants for reproducibility and configuration
RANDOM_SEED = 42
LABEL_COLUMN = "median_house_value"
PROBLEM_TYPE = "regression"

# Maximum time allocated for each model's fit operation in seconds.
# The total remaining time is 3170 seconds for 10 folds.
# This means an average of 317 seconds per fold.
# We set a conservative limit per fit to account for overhead (loading,
# preprocessing, prediction, and potential variations in execution time).
FIT_TIME_LIMIT_SECONDS = 250

class AutoGluonMultiModalWrapper:
    """
    A wrapper class for AutoGluon MultiModalPredictor to conform to the
    fit/predict interface required by the evaluation harness.
    """
    def __init__(self):
        self.predictor = None
        self.model_path = None # Path where the model artifacts will be stored
        self.label_column = LABEL_COLUMN

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon MultiModalPredictor model.

        All preprocessing and model fitting are performed within this method,
        using only the provided training data (X, y).
        A unique temporary directory is created for each fit to ensure
        "fresh artifact directories per fold".

        Args:
            X (pd.DataFrame): Training features.
            y (pd.Series): Training target values.
        """
        # Create a unique temporary directory for each fold's model artifacts.
        # This ensures "fresh artifact directories per fold" as required.
        # The harness is expected to clean up these directories.
        model_dir_name = f"autogluon_multimodal_model_{uuid.uuid4().hex}"
        self.model_path = Path(model_dir_name)
        self.model_path.mkdir(parents=True, exist_ok=True)

        # Combine X and y into a single DataFrame, as MultiModalPredictor
        # expects the label column to be part of the training DataFrame.
        train_data = X.copy()
        train_data[self.label_column] = y

        # Initialize MultiModalPredictor.
        # problem_type is explicitly set to "regression".
        self.predictor = MultiModalPredictor(
            label=self.label_column,
            problem_type=PROBLEM_TYPE,
            path=str(self.model_path) # MultiModalPredictor expects a string path
        )

        # Fit the predictor.
        # The random_seed is passed to the fit method for reproducibility,
        # as per AutoGluon's API for this version.
        # time_limit is set to ensure fits complete within the overall deadline.
        self.predictor.fit(
            train_data=train_data,
            time_limit=FIT_TIME_LIMIT_SECONDS,
            random_seed=RANDOM_SEED,
            # presets="best_quality" can be added for potentially better performance
            # but might increase training time. Default presets are often a good balance.
        )

    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions using the fitted AutoGluon MultiModalPredictor model.

        Args:
            X (pd.DataFrame): Features for which to make predictions.

        Returns:
            np.ndarray: An array of predicted numerical (float) values.
        """
        if self.predictor is None:
            raise RuntimeError("Model has not been fitted yet. Call .fit(X, y) first.")
        
        # Predict on the new data.
        # MultiModalPredictor.predict for regression returns a pandas Series.
        # We convert it to a numpy array as typically expected for prediction outputs.
        predictions = self.predictor.predict(X)
        return predictions.values

def build_model(X_train: pd.DataFrame, y_train: pd.Series) -> AutoGluonMultiModalWrapper:
    """
    Factory function to generate a fresh, unfitted model wrapper.

    This function is called by the harness for each fold of cross-validation.
    It must return a new instance of the model wrapper, ready to be fitted.

    Args:
        X_train (pd.DataFrame): Training features (not used for initialization,
                                but provided by the contract).
        y_train (pd.Series): Training target (not used for initialization,
                              but provided by the contract).

    Returns:
        AutoGluonMultiModalWrapper: An unfitted model wrapper instance
                                    with .fit(X,y) and .predict(X) methods.
    """
    return AutoGluonMultiModalWrapper()
```
