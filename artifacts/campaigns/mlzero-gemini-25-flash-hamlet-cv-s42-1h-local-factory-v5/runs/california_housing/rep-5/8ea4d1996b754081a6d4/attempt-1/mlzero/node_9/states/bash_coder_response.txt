```bash
#!/bin/bash
set -euo pipefail

# Define the target conda environment directory for node_9
ENV_DIR="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/california_housing/rep-5/8ea4d1996b754081a6d4/attempt-1/mlzero/node_9/conda_env"

# Define paths to the requirements files
REQ_FILE_COMMON="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
REQ_FILE_TABULAR="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"

echo "--- Starting Conda Environment Setup for node_9 ---"

# 1. Create the conda environment with Python 3.11
echo "Creating conda environment at '$ENV_DIR' with Python 3.11..."
conda create --prefix "$ENV_DIR" python=3.11 -y || { echo "ERROR: Failed to create conda environment at '$ENV_DIR'."; exit 1; }

# 2. Activate the newly created environment
echo "Activating conda environment..."
# Ensure conda is initialized in the current shell for `conda activate` to work.
# This is a robust way to activate in non-interactive scripts, addressing CONDA_SHLVL issues.
if [ -z "${CONDA_BASE_PATH:-}" ]; then
    CONDA_BASE=$(conda info --base 2>/dev/null || echo "/opt/conda") # Fallback if conda info fails
    if [ -f "$CONDA_BASE/etc/profile.d/conda.sh" ]; then
        source "$CONDA_BASE/etc/profile.d/conda.sh"
    else
        echo "ERROR: conda.sh not found. Ensure Conda is installed and initialized correctly."
        exit 1
    fi
fi
# Safely check CONDA_SHLVL to prevent unbound variable errors with 'set -u'
if [ -z "${CONDA_SHLVL:-}" ]; then
    echo "WARNING: CONDA_SHLVL not set, attempting to activate without it. This might indicate an unusual shell environment."
fi
conda activate "$ENV_DIR" || { echo "ERROR: Failed to activate conda environment at '$ENV_DIR'."; exit 1; }

# 3. Install uv using the environment's pip
echo "Installing uv into the environment..."
pip install uv || { echo "ERROR: Failed to install uv into the environment."; exit 1; }

# 4. Install required packages from the specified requirements files using uv.
# Explicitly include numpy, pandas, and scikit-learn as requested,
# along with packages from the requirements files. uv will handle dependencies,
# version resolution, and avoid unnecessary upgrades/reinstalls.
echo "Installing numpy, pandas, scikit-learn, and packages from requirements files using uv pip install..."
uv pip install numpy pandas scikit-learn -r "$REQ_FILE_TABULAR" --prerelease=allow -r "$REQ_FILE_COMMON" || { echo "ERROR: Failed to install required packages using uv."; exit 1; }

echo "--- Conda Environment Setup Complete for node_9 ---"
```
