import pandas as pd
import numpy as np
import os
import uuid
import logging
from autogluon.tabular import TabularPredictor, TabularDataset

# --- Module-level constants and configurations ---

# Target column name for the regression task.
TARGET_COLUMN = 'median_house_value'

# Random seed for reproducibility across AutoGluon's internal operations.
RANDOM_SEED = 42

# Time limit in seconds for each individual model fit operation.
# The total remaining search time is 903 seconds for ALL ten fits.
# Allocating 80 seconds per fit (80 * 10 = 800 seconds) leaves
# 103 seconds for overhead (data loading, Python startup, etc.) across all folds,
# ensuring all fits finish within the shared deadline.
FIT_TIME_LIMIT_SECONDS = 80

# Configure logging to suppress excessive AutoGluon output.
# This helps keep benchmark logs clean and focused.
logging.basicConfig(level=logging.WARNING)
logging.getLogger('autogluon').setLevel(logging.WARNING)
logging.getLogger('autogluon.core').setLevel(logging.WARNING)
logging.getLogger('autogluon.tabular').setLevel(logging.WARNING)


class AutoGluonTabularWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor to conform to the
    fit/predict interface required by the evaluation harness.

    This wrapper ensures that:
    1. A fresh, unfitted model is returned by `build_model`.
    2. All preprocessing and model fitting occur within the `fit` method,
       using only the training data provided to that method.
    3. Unique artifact directories are used for each model instance (per fold).
    4. The `fit` operation is time-bound.
    """

    def __init__(self):
        """
        Initializes the AutoGluon TabularPredictor with a unique path for artifacts
        and a random seed for reproducibility.
        """
        # Create a unique path for AutoGluon artifacts for each model instance.
        # This ensures that each fold's model has its own isolated storage,
        # preventing conflicts and ensuring "fresh artifact directories per fold".
        # The 'autogluon_artifacts' directory will be created in the current working directory
        # of the execution environment.
        self.model_path = os.path.join('autogluon_artifacts', str(uuid.uuid4()))

        # Initialize TabularPredictor.
        # - `label`: Specifies the target column.
        # - `path`: Directory to store model artifacts.
        # - `problem_type`: Explicitly set to 'regression' for this task.
        # - `eval_metric`: Set to 'rmse' as it's a common metric for regression
        #                  and AutoGluon's default for regression.
        # - `verbosity=0`: Suppresses most AutoGluon output during training
        #                  for cleaner benchmark logs.
        # - `random_seed`: Sets the seed for reproducibility. This is the correct
        #                  place for the seed in AutoGluon 1.6.x versions.
        self.predictor = TabularPredictor(
            label=TARGET_COLUMN,
            path=self.model_path,
            problem_type='regression',
            eval_metric='rmse',
            verbosity=0,
            random_seed=RANDOM_SEED  # Set random seed for reproducibility
        )
        self.is_fitted = False

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon TabularPredictor model using the provided training data.
        All learned preprocessing and model training are performed within this method.

        Args:
            X (pd.DataFrame): Training features.
            y (pd.Series): Training target values.
        """
        # Combine features (X) and target (y) into a single DataFrame,
        # which is the expected input format for TabularPredictor's `fit` method.
        train_df = X.copy()
        train_df[TARGET_COLUMN] = y

        # Convert the pandas DataFrame to AutoGluon's TabularDataset format.
        train_data = TabularDataset(train_df)

        # Fit the predictor.
        # - `train_data`: The combined training features and target.
        # - `time_limit`: Bounds the training time for this fold, ensuring
        #                 all fits finish within the shared deadline.
        # - `presets='best'`: Uses AutoGluon's configuration for high accuracy,
        #                     which includes various models and ensembling.
        # AutoGluon automatically handles internal validation splits and
        # feature engineering within this `fit` call.
        self.predictor.fit(
            train_data=train_data,
            time_limit=FIT_TIME_LIMIT_SECONDS,
            presets='best',
        )
        self.is_fitted = True

    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions using the fitted AutoGluon TabularPredictor model.

        Args:
            X (pd.DataFrame): Features for which to make predictions.

        Returns:
            np.ndarray: An array of predicted numerical (float) values.
        """
        if not self.is_fitted:
            raise RuntimeError("Model has not been fitted yet. Call .fit(X, y) first.")

        # AutoGluon's predict method directly takes a DataFrame of features.
        predictions = self.predictor.predict(X)

        # Ensure the output is a numpy array of float values, as required by the contract.
        return predictions.values.astype(float)
