```bash
#!/bin/bash
set -e

# Define environment and file paths
CONDA_ENV_DIR="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/wilt/rep-2/f3676860d054e0e4d222/attempt-1/mlzero/node_6/conda_env"
COMMON_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
MULTIMODAL_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.multimodal/requirements.txt"

echo "--- Starting Conda environment setup for node_6 ---"

# 1. Check if conda is available
command -v conda &> /dev/null || { echo "Error: conda command not found. Please ensure Anaconda or Miniconda is installed and in your PATH."; exit 1; }

# 2. Create the conda environment with Python 3.11
# Remove existing environment directory to ensure a clean installation
if [ -d "$CONDA_ENV_DIR" ]; then
    echo "Removing existing environment directory: $CONDA_ENV_DIR"
    rm -rf "$CONDA_ENV_DIR"
fi
echo "Creating conda environment at $CONDA_ENV_DIR with Python 3.11..."
conda create -p "$CONDA_ENV_DIR" python=3.11 -y

# 3. Activate the environment
echo "Activating conda environment..."
# Source conda.sh to make 'conda activate' available in the script's shell
source "$(conda info --base)/etc/profile.d/conda.sh"
conda activate "$CONDA_ENV_DIR"
echo "Conda environment activated: $(which python)"

# 4. Install uv using the environment's pip
echo "Installing uv into the environment..."
pip install uv

# 5. Install required packages using uv
echo "Installing numpy, pandas, scikit-learn, and packages from requirements files using uv..."
uv pip install \
    numpy pandas scikit-learn \
    -r "$MULTIMODAL_REQUIREMENTS_FILE" \
    --prerelease=allow \
    -r "$COMMON_REQUIREMENTS_FILE"

echo "--- Conda environment setup complete for node_6 ---"
echo "Environment is ready at $CONDA_ENV_DIR"
echo "Python interpreter: $(which python)"
echo "Installed packages:"
uv pip list
```
