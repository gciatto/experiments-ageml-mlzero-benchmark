class AutoGluonMultiModalWrapper:
    """
    A wrapper class for AutoGluon MultiModalPredictor to conform to the
    fit/predict interface required by the evaluation harness.
    """
    def __init__(self, problem_type: str, random_seed: int, time_limit_per_fit: int):
        """
        Initializes the wrapper. The actual MultiModalPredictor is initialized
        and fitted within the `fit` method to ensure a fresh instance per fold.

        Args:
            problem_type (str): The type of machine learning problem (e.g., 'classification').
            random_seed (int): Seed for reproducibility.
            time_limit_per_fit (int): Maximum time in seconds allowed for each fit call.
        """
        self.problem_type = problem_type
        self.random_seed = random_seed
        self.time_limit_per_fit = time_limit_per_fit
        self.predictor: MultiModalPredictor = None
        self.label_column_name: str = None
        self.artifact_path: Path = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon MultiModalPredictor using the provided training data.

        Args:
            X (pd.DataFrame): Training features.
            y (pd.Series): Training labels.
        """
        self.label_column_name = y.name
        if self.label_column_name is None:
            # If y is an unnamed Series, give it a default name
            self.label_column_name = "target"
            y = y.rename(self.label_column_name)

        # Combine X and y into a single DataFrame as required by MultiModalPredictor
        train_data = pd.concat([X, y], axis=1)

        # Create a unique path for artifacts for this fold.
        # This ensures fresh artifact directories per fold and avoids conflicts.
        self.artifact_path = Path(f"./autogluon_artifacts_{uuid.uuid4().hex}")
        self.artifact_path.mkdir(parents=True, exist_ok=True)  # Ensure directory exists

        # Initialize MultiModalPredictor.
        # Using 'fast_inference' preset to prioritize speed, which is crucial
        # for completing all ten folds within the shared deadline.
        self.predictor = MultiModalPredictor(
            label=self.label_column_name,
            problem_type=self.problem_type,
            path=str(self.artifact_path),
        )

        # Fit the predictor.
        # The `seed` argument in `fit` ensures reproducibility for this specific run.
        # `hyperparameters={"env.log_level": "warning"}` helps suppress verbose output.
        self.predictor.fit(
            train_data=train_data,
            time_limit=self.time_limit_per_fit,
            presets='fast_inference',
            seed=self.random_seed,
            hyperparameters={"env.log_level": "warning"}
        )

    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions on new data using the fitted MultiModalPredictor.

        Args:
            X (pd.DataFrame): Features for which to make predictions.

        Returns:
            np.ndarray: A 1D array of predicted class labels.
        """
        if self.predictor is None:
            raise RuntimeError("Model has not been fitted yet. Call .fit(X, y) first.")
        
        # MultiModalPredictor.predict returns a pandas Series, convert to 1D numpy array.
        predictions = self.predictor.predict(X)
        return predictions.to_numpy()

    def __del__(self):
        """
        Clean up the artifact directory when the wrapper object is deleted.
        This helps manage disk space, especially in environments with many folds.
        """
        if self.artifact_path and self.artifact_path.exists():
            try:
                shutil.rmtree(self.artifact_path)
            except OSError as e:
                logging.warning(f"Error removing artifact directory {self.artifact_path}: {e}")
