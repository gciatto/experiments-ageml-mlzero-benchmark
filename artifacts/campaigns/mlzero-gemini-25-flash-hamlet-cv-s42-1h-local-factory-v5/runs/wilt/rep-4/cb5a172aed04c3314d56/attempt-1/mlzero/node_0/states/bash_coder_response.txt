```bash
#!/bin/bash
set -euo pipefail

# Define environment and file paths
ENV_BASE_DIR="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/wilt/rep-4/cb5a172aed04c3314d56/attempt-1/mlzero/node_0"
ENV_DIR="${ENV_BASE_DIR}/conda_env"
COMMON_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
AUTOGLUON_TABULAR_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"

echo "Starting conda environment setup for AutoGluonTabularModel..."
echo "Environment path: ${ENV_DIR}"

# Check if conda is available and initialize it for the current shell session
if ! command -v conda &> /dev/null; then
    echo "Error: 'conda' command not found. Please ensure Conda is installed and in your PATH."
    exit 1
fi

# Initialize conda for the current shell session if not already done.
# This makes 'conda activate' available.
# Check if the base conda init script exists and source it.
if [ -f "$(conda info --base)/etc/profile.d/conda.sh" ]; then
    source "$(conda info --base)/etc/profile.d/conda.sh"
else
    echo "Error: conda initialization script not found at expected location: $(conda info --base)/etc/profile.d/conda.sh"
    echo "Cannot activate environment. Please check your conda installation."
    exit 1
fi

# Remove existing environment if it exists to ensure a clean slate
if conda env list | grep -q "${ENV_DIR}"; then
    echo "Removing existing conda environment at ${ENV_DIR}..."
    conda env remove -p "${ENV_DIR}" -y
fi

echo "Creating conda environment with Python 3.11 at ${ENV_DIR}..."
conda create -p "${ENV_DIR}" python=3.11 -y

echo "Activating conda environment: ${ENV_DIR}"
conda activate "${ENV_DIR}"

echo "Installing uv..."
pip install uv

echo "Installing required packages from requirement files using uv..."
# Install packages from the specified requirement files.
# The order is important as specified in the prompt: autogluon.tabular requirements first, then common.
# This command will pull in autogluon.tabular, autogluon.core, pandas, numpy, scikit-learn,
# and all other necessary dependencies as specified in the requirement files.
uv pip install -r "${AUTOGLUON_TABULAR_REQ_FILE}" --prerelease=allow -r "${COMMON_REQ_FILE}"

echo "Conda environment setup complete."
```
