```bash
#!/bin/bash
set -euo pipefail

# Define environment and file paths
ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/wilt/rep-5/0cf989f159b35f94ca0e/attempt-1/mlzero/node_2/conda_env"
COMMON_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
ML_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/machine learning/requirements.txt"

echo "--- Starting Conda Environment Setup ---"

# 1. Create the conda environment with Python 3.11
echo "Creating conda environment at $ENV_PATH with Python 3.11..."
conda create -p "$ENV_PATH" python=3.11 -y

# 2. Activate the environment
echo "Activating conda environment..."
# Initialize conda for the current shell session if not already done
# This line is crucial for `conda activate` to work in non-interactive scripts
source "$(conda info --base)/etc/profile.d/conda.sh"
conda activate "$ENV_PATH"

echo "Conda environment activated: $(which python)"

# 3. Install uv
echo "Installing uv into the environment..."
pip install uv

# 4. Install required packages from requirements.txt files using uv
echo "Installing packages from requirements.txt files using uv..."
uv pip install -r "$ML_REQ_FILE" --prerelease=allow -r "$COMMON_REQ_FILE"

# 5. Install additional packages: numpy, pandas, scikit-learn, and autogluon.tabular
echo "Installing numpy, pandas, scikit-learn, and autogluon.tabular using uv..."
uv pip install numpy pandas scikit-learn autogluon.tabular

echo "--- Conda Environment Setup Complete ---"
```
