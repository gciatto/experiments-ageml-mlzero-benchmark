TARGET_COLUMN = 'class'
PROBLEM_TYPE = 'regression'
# The problem description specifies 'rmse (lower is better)'.
# AutoGluon's equivalent is 'root_mean_squared_error'.
EVAL_METRIC = 'root_mean_squared_error'
RANDOM_SEED = 42

# The total shared deadline is 1775 seconds for all ten fits.
# To ensure all fits complete, we allocate a conservative time limit per fold.
# 1775 seconds / 10 folds = 177.5 seconds per fold.
# We set it slightly lower to account for overhead and potential variability.
TIME_LIMIT_PER_FOLD = 150  # seconds

# As per guidance, we try to use presets='extreme'.
# Note: 'extreme' preset often requires a GPU for optimal performance and
# is designed for datasets with <30K samples. Our dataset is small (398 rows).
# If a GPU is not available, AutoGluon might fall back to CPU, which could be
# significantly slower and potentially hit the time limit.
PRESETS = 'extreme'

class AutoGluonTabularWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor to conform to the
    fit/predict interface required by the benchmark harness.
    """
    def __init__(self):
        self.predictor = None
        self.path = None  # Path for AutoGluon artifacts for this specific model instance
        self.is_fitted = False

    def fit(self, X_train: pd.DataFrame, y_train: pd.Series):
        """
        Fits the AutoGluon TabularPredictor model.

        All preprocessing and model fitting are performed within this method,
        using only the provided training data. A fresh, temporary directory
        is created for AutoGluon artifacts to ensure isolation per fold.

        Args:
            X_train (pd.DataFrame): Training features.
            y_train (pd.Series): Training target values.
        """
        # Set seeds for reproducibility for this specific fit call
        # The autogluon.core.utils.utils.set_seed or autogluon.common.utils.init_utils.set_seed
        # function has been problematic across AutoGluon versions.
        # We rely on numpy and random for general Python seeding, and pass
        # random_seed directly to TabularPredictor for AutoGluon's internal seeding.
        np.random.seed(RANDOM_SEED)
        random.seed(RANDOM_SEED)
        os.environ['PYTHONHASHSEED'] = str(RANDOM_SEED)

        # Create a temporary directory for AutoGluon artifacts for this fold.
        # This ensures "fresh artifact directories per fold" as required.
        self.path = tempfile.mkdtemp(prefix="autogluon_model_")

        # Combine X_train and y_train into a single DataFrame for AutoGluon.
        # AutoGluon expects the target column to be part of the training data.
        train_data = X_train.copy()
        train_data[TARGET_COLUMN] = y_train

        # Initialize TabularPredictor with specified problem type, evaluation metric,
        # the unique path for artifacts, and the random_seed for reproducibility.
        self.predictor = TabularPredictor(
            label=TARGET_COLUMN,
            problem_type=PROBLEM_TYPE,
            eval_metric=EVAL_METRIC,
            path=self.path,
            random_seed=RANDOM_SEED, # Pass seed directly to TabularPredictor
        )
        
        # Fit the predictor with the specified time limit and presets.
        self.predictor.fit(
            train_data=TabularDataset(train_data),
            time_limit=TIME_LIMIT_PER_FOLD,
            presets=PRESETS,
        )
        self.is_fitted = True

    def predict(self, X_test: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions using the fitted AutoGluon TabularPredictor model.

        Args:
            X_test (pd.DataFrame): Features for which to make predictions.

        Returns:
            np.ndarray: Predicted continuous numerical values.
        """
        if not self.is_fitted:
            raise RuntimeError("Model is not fitted yet. Call fit() before predict().")
        
        # AutoGluon's predict method expects a TabularDataset or pandas DataFrame.
        test_data = TabularDataset(X_test.copy())
        
        predictions = self.predictor.predict(test_data)
        
        # Return predictions as a numpy array as required by the contract.
        return predictions.values

    def _cleanup(self):
        """
        Cleans up the temporary directory created by AutoGluon.
        This method is not explicitly called by the benchmark harness but is good practice.
        """
        if self.path and os.path.exists(self.path):
            try:
                shutil.rmtree(self.path)
            except OSError as e:
                # Log or print error if cleanup fails, but don't block execution.
                print(f"Error cleaning up AutoGluon path {self.path}: {e}")
            self.path = None

# --- Factory function required by the benchmark contract ---
def build_model(X_train: pd.DataFrame, y_train: pd.Series):
    """
    Builds and returns a fresh, unfitted model wrapper.

    This function serves as the factory for creating new model instances
    for each fold of the cross-validation.

    Args:
        X_train (pd.DataFrame): Training features (used for type hinting, not for model initialization).
        y_train (pd.Series): Training target (used for type hinting, not for model initialization).

    Returns:
        AutoGluonTabularWrapper: An unfitted model wrapper with `fit(X,y)` and `predict(X)` methods.
    """
    return AutoGluonTabularWrapper()

# Automatic Feature Engineering in AutoGluon

## Feature Processing by Data Type

### Numerical Columns
- No automated feature engineering for integer and floating point columns

### Categorical Columns
- Mapped to monotonically increasing integers for model compatibility

### Datetime Columns
- Converted to numerical Pandas datetime values
- Extracted features: `[year, month, day, dayofweek]` by default
- Missing/invalid values replaced with mean
- Note: Pandas datetime has min/max limits that may affect extreme dates

### Text Columns
- With MultiModal option: processed using Transformer neural networks
- Without MultiModal:
  - N-gram feature generation (creates "n-hot" encoded columns)
  - Special features (word counts, character counts, etc.)

## Additional Processing
- Single-value columns are dropped
- Duplicate columns are removed

## Implementation Example

```python
# Basic usage
predictor = TabularPredictor(label='class', problem_type='multiclass').fit(train_data)

# Custom feature generator
from autogluon.features.generators import PipelineFeatureGenerator, CategoryFeatureGenerator, IdentityFeatureGenerator
from autogluon.common.features.types import R_INT, R_FLOAT

mypipeline = PipelineFeatureGenerator(
    generators = [[        
        CategoryFeatureGenerator(maximum_num_cat=10),  # Limit categorical values
        IdentityFeatureGenerator(infer_features_in_args=dict(valid_raw_types=[R_INT, R_FLOAT])),
    ]]
)

# Use custom feature generator with predictor

predictor.fit(df, hyperparameters={'GBM': {}}, feature_generator=mypipeline)
```

## Missing Value Handling
- Datetime columns: missing values replaced with mean
- Other columns: NaN values preserved

## Customization Tips
- Use `PipelineFeatureGenerator` for custom pipelines
- Configure individual generators with parameters like `maximum_num_cat`
- For categorical recognition, explicitly set column type: `df["col"] = df["col"].astype("category")`

```python
ag_accuracy_scorer = make_scorer(

    greater_is_better=True,

)
```

### Custom Mean Squared Error Metric

```python
ag_mean_squared_error_scorer = make_scorer(

)
```

### Custom Function Example
```python
def mse_func(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    return ((y_true - y_pred) ** 2).mean()

ag_mean_squared_error_custom_scorer = make_scorer(
    name='mean_squared_error',
    score_func=mse_func,
    optimum=0,
    greater_is_better=False
```

### Custom ROC AUC Metric
```python
ag_roc_auc_scorer = make_scorer(
    name='roc_auc',
    score_func=sklearn.metrics.roc_auc_score,
    optimum=1,
    greater_is_better=True,
    needs_threshold=True
)
```

## Key Parameters for `make_scorer`
- `name`: Name for the scorer (used in logs)
- `score_func`: The function to wrap
- `optimum`: Best possible value from the original metric
- `greater_is_better`: Whether higher values are better
- `needs_*`: Specify the type of predictions required:
  - `needs_pred`: For regression metrics (default if no others specified)
  - `needs_proba`: For probability estimates
  - `needs_class`: For classification predictions
  - `needs_threshold`: For continuous decision certainty (binary classification)
  - `needs_quantile`: For quantile regression

### With leaderboard
```python
predictor = TabularPredictor(label=label).fit(train_data, hyperparameters='toy')
```

### As evaluation metric
```python

```

## Important Notes
- AutoGluon Scorers always report scores in `greater_is_better=True` form internally
- For metrics where `greater_is_better=False`, AutoGluon flips the value
- Error is calculated as `sign * optimum - score` where `sign=1` if `greater_is_better=True`, else `sign=-1`
- Error is always in `lower_is_better` format (0 = perfect prediction)

## 1. Mitra: AutoGluon's Tabular Foundation Model

Mitra is a state-of-the-art tabular foundation model that excels on small datasets (<5,000 samples, <100 features) for both classification and regression tasks.

### Zero-Shot Classification

```python
mitra_predictor = TabularPredictor(label='target')
mitra_predictor.fit(
    wine_train_data,
    hyperparameters={
        'MITRA': {'fine_tune': False}
    }
)

# Evaluate
mitra_predictor.leaderboard(wine_test_data)
```

### Fine-tuned Classification

```python
mitra_predictor_ft = TabularPredictor(label='target')
mitra_predictor_ft.fit(
    wine_train_data,
    hyperparameters={
        'MITRA': {'fine_tune': True, 'fine_tune_steps': 10}
    },
    time_limit=120  # 2 minutes
)
```

### Regression

```python
mitra_reg_predictor = TabularPredictor(
```

## Advanced Usage: Combining Multiple Foundational Models

```python
multi_foundation_config = {
    'MITRA': {
        'fine_tune': True,
        'fine_tune_steps': 10
    },
    'TABPFNV2': {},
    'TABICL': {},
}

ensemble_predictor = TabularPredictor(
    label='target',
    path='./ensemble_foundation_model'

    wine_train_data,
    hyperparameters=multi_foundation_config,
    time_limit=300  # More time for multiple models
)
```
