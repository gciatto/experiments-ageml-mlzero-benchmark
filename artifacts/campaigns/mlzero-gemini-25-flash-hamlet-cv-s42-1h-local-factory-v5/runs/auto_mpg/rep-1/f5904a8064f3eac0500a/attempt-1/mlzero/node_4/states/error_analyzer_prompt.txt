# --- Model Wrapper Class ---
class AutoGluonModelWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor to conform to the

    def __init__(self, path_prefix: str = "AutogluonModels", time_limit: int = 200):
        """
        Initializes the wrapper. The actual TabularPredictor instance is created
        and fitted within the `fit` method to ensure it's fresh for each fold
        and uses the correct, unique artifact path.

        Args:
            path_prefix (str): A subdirectory name where AutoGluon will store its artifacts.
                               This will be created relative to the current working directory
                               of the execution environment, which is unique per fold.
            time_limit (int): The maximum time in seconds AutoGluon is allowed to train
                              for a single fold. This helps manage the total shared deadline.

        self.predictor: TabularPredictor = None
        self.path_prefix = path_prefix
        self.time_limit = time_limit
        self.model_path: str = None # Will be set dynamically in fit()

    def fit(self, X_train: pd.DataFrame, y_train: pd.Series):
        """
        Fits the AutoGluon TabularPredictor using the provided training data.
        All learned preprocessing, feature engineering, and model training
        are performed within this method, using only the `X_train` and `y_train`
        for the current fold.

        Args:
            X_train (pd.DataFrame): Training features for the current fold.
            y_train (pd.Series): Training target values for the current fold.
        """
        # Create a unique path for AutoGluon artifacts for this specific fit call.
        # The harness ensures the current working directory is unique per fold,
        # so a simple subdirectory name within it is sufficient for isolation.
        self.model_path = os.path.join(os.getcwd(), self.path_prefix)
        
        # AutoGluon's TabularPredictor expects the target column to be part of
        # the DataFrame passed to `fit`.
        train_data = X_train.copy()
        target_column = 'class' # As specified in the task description
        train_data[target_column] = y_train

        # Initialize TabularPredictor for regression.
        # `verbosity=0` suppresses most of AutoGluon's console output.
        self.predictor = TabularPredictor(
            label=target_column,
            path=self.model_path,
            problem_type='regression',

            verbosity=0
        )

        # `presets='medium_quality_fast_inference'` balances model quality with training speed,
        # which is crucial for meeting the shared deadline across multiple folds.
        self.predictor.fit(

            time_limit=self.time_limit,
            presets='medium_quality_fast_inference',
            # AutoGluon typically respects global numpy/random seeds or manages its own internal seeds.

COMMON_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
ML_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/machine learning/requirements.txt"

echo "--- Starting Conda Environment Setup ---"

# 1. Ensure parent directories exist for the conda environment
mkdir -p "$(dirname "$CONDA_ENV_PATH")"

# 2. Create Conda environment with Python 3.11
echo "Creating conda environment at $CONDA_ENV_PATH with Python 3.11..."
conda create -p "$CONDA_ENV_PATH" python=3.11 -y || { echo "Error: Failed to create conda environment."; exit 1; }

# 3. Activate the environment
echo "Activating conda environment..."
# Robust activation for scripts: source conda.sh then activate
CONDA_BASE=$(conda info --base)
if [ -f "$CONDA_BASE/etc/profile.d/conda.sh" ]; then
    source "$CONDA_BASE/etc/profile.d/conda.sh" || { echo "Error: Could not source conda.sh."; exit 1; }
else
    echo "Error: conda.sh not found at $CONDA_BASE/etc/profile.d/conda.sh. Is conda installed and in PATH?"
    exit 1
fi
conda activate "$CONDA_ENV_PATH" || { echo "Error: Could not activate conda environment."; exit 1; }

echo "Conda environment activated: $(which python)"
echo "Python version: $(python --version)"

# 4. Install uv
echo "Installing uv..."
pip install uv || { echo "Error: Failed to install uv."; exit 1; }

# 5. Install core packages: numpy, pandas, scikit-learn
echo "Installing numpy, pandas, scikit-learn using uv..."
uv pip install numpy pandas scikit-learn || { echo "Error: Failed to install core packages."; exit 1; }

# 6. Install AutoGluon (the selected ML library)
echo "Installing autogluon.tabular using uv..."
# Installing autogluon.tabular will pull in autogluon.common and other necessary components.
# This approach addresses previous errors related to AutoGluon's internal dependency management
# by letting autogluon.tabular manage its own dependencies.
uv pip install autogluon.tabular || { echo "Error: Failed to install autogluon.tabular."; exit 1; }

# 7. Install required packages from the specified requirement files using uv
echo "Installing packages from requirement files using uv..."
# The order and --prerelease=allow flag are applied as specified in the prompt.
uv pip install -r "$ML_REQ_FILE" --prerelease=allow -r "$COMMON_REQ_FILE" || { echo "Error: Failed to install packages from requirement files."; exit 1; }

echo "--- Conda Environment Setup Complete ---"
