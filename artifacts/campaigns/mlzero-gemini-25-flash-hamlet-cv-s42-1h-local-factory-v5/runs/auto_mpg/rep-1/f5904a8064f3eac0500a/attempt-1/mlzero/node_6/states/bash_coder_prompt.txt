# --- Model Wrapper Class ---
class AutoGluonModelWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor to conform to the
    `fit(X, y)` and `predict(X)` interface required by the evaluation harness.
    """
    def __init__(self, path_prefix: str = "AutogluonModels", time_limit: int = 150):
        """
        Initializes the wrapper. The actual TabularPredictor instance is created
        and fitted within the `fit` method to ensure it's fresh for each fold
        and uses the correct, unique artifact path.

        Args:
            path_prefix (str): A subdirectory name where AutoGluon will store its artifacts.
                               This will be created relative to the current working directory
                               of the execution environment, which is unique per fold.
            time_limit (int): The maximum time in seconds AutoGluon is allowed to train
                              for a single fold. This helps manage the total shared deadline.
        """
        self.predictor: TabularPredictor = None
        self.path_prefix = path_prefix
        self.time_limit = time_limit
        self.model_path: str = None # Will be set dynamically in fit()

    def fit(self, X_train: pd.DataFrame, y_train: pd.Series):
        """
        Fits the AutoGluon TabularPredictor using the provided training data.
        All learned preprocessing, feature engineering, and model training
        are performed within this method, using only the `X_train` and `y_train`
        for the current fold.

        Args:
            X_train (pd.DataFrame): Training features for the current fold.
            y_train (pd.Series): Training target values for the current fold.
        """
        # Create a unique path for AutoGluon artifacts for this specific fit call.
        # The harness ensures the current working directory is unique per fold,
        # so a simple subdirectory name within it is sufficient for isolation.
        self.model_path = os.path.join(os.getcwd(), self.path_prefix)
        
        # AutoGluon's TabularPredictor expects the target column to be part of
        # the DataFrame passed to `fit`.
        train_data = X_train.copy()
        target_column = 'class' # As specified in the task description
        train_data[target_column] = y_train

        # Initialize TabularPredictor for regression.
        # `verbosity=0` suppresses most of AutoGluon's console output.
        self.predictor = TabularPredictor(
            label=target_column,
            path=self.model_path,
            problem_type='regression',
            eval_metric='rmse', # Metric specified in README.md
            verbosity=0
        )

        # Fit the predictor.
        # `presets='medium_quality'` is used as per suggested fix for invalid preset.
        # `seed=SEED` is passed directly to fit for reproducibility, addressing the
        # invalid `random_seed` argument in the constructor.
        self.predictor.fit(
            train_data=train_data,
            time_limit=self.time_limit,
            presets='medium_quality',
            seed=SEED
        )

    def predict(self, X_test: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions on new data using the fitted AutoGluon TabularPredictor.

        Args:
            X_test (pd.DataFrame): Features of the instances to make predictions for.

        Returns:
            np.ndarray: An array of predicted continuous numerical values.
        """
        if self.predictor is None:
            raise RuntimeError("Model has not been fitted yet. Call .fit() first.")
        
        # AutoGluon's predict method expects a DataFrame.
        predictions = self.predictor.predict(X_test)
        
        # Return predictions as a numpy array to conform to common scikit-learn interfaces.
        return predictions.values
