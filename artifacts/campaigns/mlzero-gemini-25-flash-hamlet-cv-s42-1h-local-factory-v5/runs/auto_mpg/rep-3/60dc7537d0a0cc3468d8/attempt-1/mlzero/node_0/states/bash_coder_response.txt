```bash
#!/bin/bash
set -euo pipefail

# Define paths for the Conda environment and requirements files
CONDA_ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/auto_mpg/rep-3/60dc7537d0a0cc3468d8/attempt-1/mlzero/node_0/conda_env"
COMMON_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
AUTOGLUON_TABULAR_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"

echo "Starting Conda environment setup at $CONDA_ENV_PATH"

# Initialize Conda for the current shell session if not already done.
# This is necessary for 'conda activate' to function correctly.
# We assume the 'conda' command is available in PATH.
if [ -z "${CONDA_SHLVL:-}" ]; then # Check if conda is already initialized
    # Attempt to source conda.sh from the base conda installation
    source "$(conda info --base)/etc/profile.d/conda.sh" || { echo "ERROR: Failed to initialize Conda. Ensure Conda is installed and in your PATH." >&2; exit 1; }
fi

# Remove existing environment if it exists to ensure a clean installation
if [ -d "$CONDA_ENV_PATH" ]; then
    echo "Existing environment found at $CONDA_ENV_PATH. Removing it..."
    conda env remove --prefix "$CONDA_ENV_PATH" --yes || { echo "ERROR: Failed to remove existing conda environment." >&2; exit 1; }
fi

# Create the conda environment with Python 3.11
echo "Creating conda environment at $CONDA_ENV_PATH with Python 3.11..."
conda create --prefix "$CONDA_ENV_PATH" python=3.11 --yes || { echo "ERROR: Failed to create conda environment." >&2; exit 1; }
echo "Conda environment created."

# Activate the environment
echo "Activating conda environment..."
conda activate "$CONDA_ENV_PATH" || { echo "ERROR: Failed to activate conda environment." >&2; exit 1; }
echo "Conda environment activated."

# Install uv
echo "Installing uv..."
pip install uv || { echo "ERROR: Failed to install uv." >&2; exit 1; }
echo "uv installed."

# Install required packages from requirements files using uv
# The order of requirements files is specified in the prompt.
echo "Installing packages from requirements files using uv..."
uv pip install -r "$AUTOGLUON_TABULAR_REQUIREMENTS_FILE" --prerelease=allow -r "$COMMON_REQUIREMENTS_FILE" || { echo "ERROR: Failed to install requirements with uv." >&2; exit 1; }
echo "Requirements from files installed."

# Install numpy, pandas, and scikit-learn as explicitly requested
echo "Installing numpy, pandas, scikit-learn..."
uv pip install numpy pandas scikit-learn || { echo "ERROR: Failed to install core ML libraries." >&2; exit 1; }
echo "Core ML libraries installed."

echo "Conda environment setup complete."
```
