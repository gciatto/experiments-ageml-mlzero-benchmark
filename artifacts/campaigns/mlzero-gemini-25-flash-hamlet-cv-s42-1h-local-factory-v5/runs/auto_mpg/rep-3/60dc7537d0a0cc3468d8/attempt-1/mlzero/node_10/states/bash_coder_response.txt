```bash
#!/bin/bash
set -euo pipefail

# Define environment path and requirement file paths
CONDA_ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/auto_mpg/rep-3/60dc7537d0a0cc3468d8/attempt-1/mlzero/node_10/conda_env"
COMMON_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
ML_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/machine learning/requirements.txt"

# 1. Create the conda environment with Python 3.11
echo "Creating conda environment at $CONDA_ENV_PATH with Python 3.11..."
conda create -p "$CONDA_ENV_PATH" python=3.11 -y

# 2. Activate the environment
echo "Activating the conda environment..."
# Initialize conda for the current shell session if not already done
# This line is crucial for non-interactive scripts to use 'conda activate'
source "$(conda info --base)/etc/profile.d/conda.sh"
conda activate "$CONDA_ENV_PATH"

# 3. Install uv
echo "Installing uv..."
pip install uv

# 4. Install required packages from files and additional libraries using uv
echo "Installing packages from requirements files and additional libraries (numpy, pandas, scikit-learn, autogluon.tabular) using uv..."
uv pip install \
    -r "$ML_REQ_FILE" \
    --prerelease=allow \
    -r "$COMMON_REQ_FILE" \
    numpy pandas scikit-learn autogluon.tabular

echo "Conda environment setup complete."
```
