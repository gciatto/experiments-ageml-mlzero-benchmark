import os
import logging
from pathlib import Path

from autogluon.tabular import TabularPredictor, TabularDataset

AUTOGLUON_MODEL_DIR_BASE = 'autogluon_models'

RANDOM_SEED = 42

# The name of the target column as specified in the problem description.
TARGET_COLUMN = 'Class'

# The evaluation metric to optimize for, as specified in the problem description.

EVAL_METRIC = 'balanced_accuracy'

# Maximum time in seconds allowed for each AutoGluon fit operation.

TIME_LIMIT_PER_FOLD_SECONDS = 300

# AutoGluon presets for model quality.

# and it balances performance with training time. 'extreme' is often slower and
# may require GPU resources.
PRESETS = 'best_quality'

# --- Wrapper Class for AutoGluon TabularPredictor ---
class AutoGluonTabularWrapper:
    """
    A wrapper for AutoGluon TabularPredictor to conform to the build_model contract.
    It provides `fit(X, y)` and `predict(X)` methods.
    """
    def __init__(self, path: str, random_seed: int, time_limit: int):

                        This path should be unique for each cross-validation fold.
            random_seed (int): The random seed to use for AutoGluon's internal processes.
            time_limit (int): The maximum time in seconds allowed for the `fit` operation.
        """
        self.path = path
        self.random_seed = random_seed
        self.time_limit = time_limit
        self.predictor = None  # The actual TabularPredictor instance, initialized during fit()

        # Configuration parameters for initializing TabularPredictor.
        self.predictor_init_kwargs = {
            'label': TARGET_COLUMN,
            'eval_metric': EVAL_METRIC,
            'path': self.path,
            'verbosity': 0  # Suppress AutoGluon's detailed output during training
        }

        # Configuration parameters for the `fit()` method of TabularPredictor.
        self.fit_kwargs = {
            'presets': PRESETS,
            'time_limit': self.time_limit,
            'seed': self.random_seed,
            # For binary classification tasks, especially when optimizing for metrics
            # like 'balanced_accuracy', calibrating the decision threshold can improve
            # performance. AutoGluon handles this automatically when set to True.
            'calibrate_decision_threshold': True,
        }

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon TabularPredictor model using the provided training data.
        All preprocessing and model training are performed within this method.

        Args:
            X (pd.DataFrame): Training features.
            y (pd.Series): Training target labels.
        """
        # AutoGluon's `fit` method expects the target column to be present
        # within the training DataFrame.
        train_data = X.copy()
        train_data[TARGET_COLUMN] = y

        # Initialize TabularPredictor. This is done here to ensure a fresh instance
        # for each `fit` call, adhering to the "unfitted" model contract.
        self.predictor = TabularPredictor(**self.predictor_init_kwargs)
        
        # Convert the combined DataFrame to a TabularDataset and fit the model.
        self.predictor.fit(train_data=TabularDataset(train_data), **self.fit_kwargs)

    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions using the fitted AutoGluon TabularPredictor model.

        Args:
            X (pd.DataFrame): Features for which to make predictions.

        Returns:
            np.ndarray: A 1-dimensional numpy array of predicted class labels.
        """
        if self.predictor is None:
            raise RuntimeError("Model has not been fitted yet. Call .fit(X, y) first.")

        # Convert input features to a TabularDataset for prediction.
        test_data = TabularDataset(X.copy())
        
        # Get predictions. AutoGluon's `predict` method typically returns a pandas Series.
        predictions_series = self.predictor.predict(test_data)
        
        # Convert the predictions to a 1-dimensional numpy array, as required by the contract.
        return predictions_series.to_numpy()
