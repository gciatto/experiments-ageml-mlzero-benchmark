```bash
#!/bin/bash
set -euxo pipefail

# Define the full path for the conda environment
# The exact candidate environment path is provided in the prompt.
ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/madelon/rep-4/3f1d9fe8ce5e73e90c19/attempt-1/mlzero/node_3/conda_env"

# Define paths to the requirements files
COMMON_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
ML_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/machine learning/requirements.txt"

# Ensure the parent directory for the conda environment exists
mkdir -p "$(dirname "${ENV_PATH}")"

# 1. Create the conda environment with Python 3.11
echo "Creating conda environment at ${ENV_PATH} with Python 3.11..."
conda create -p "${ENV_PATH}" python=3.11 -y || { echo "Error: Failed to create conda environment at ${ENV_PATH}."; exit 1; }

# 2. Activate the environment
echo "Activating conda environment..."
# Initialize conda for the current shell if not already done
# This is crucial for `conda activate` to work reliably in non-interactive scripts
source "$(conda info --base)/etc/profile.d/conda.sh" || { echo "Error: Could not source conda.sh. Is conda installed and in PATH?"; exit 1; }
conda activate "${ENV_PATH}" || { echo "Error: Could not activate conda environment at ${ENV_PATH}. Check if 'bin/activate' exists."; exit 1; }

# 3. Install uv
echo "Installing uv into the environment..."
# Use the activated environment's pip
pip install uv || { echo "Error: Failed to install uv."; exit 1; }

# 4. Install required packages from the specified requirements files using uv
echo "Installing packages from requirements files using uv..."
uv pip install -r "${ML_REQUIREMENTS_FILE}" --prerelease=allow -r "${COMMON_REQUIREMENTS_FILE}" || { echo "Error: Failed to install packages from requirements files."; exit 1; }

# 5. Install additional packages: numpy, pandas, scikit-learn, and autogluon.tabular
# These are explicitly requested and derived from the factory module's imports.
echo "Installing numpy, pandas, scikit-learn, and autogluon.tabular..."
uv pip install numpy pandas scikit-learn autogluon.tabular || { echo "Error: Failed to install additional packages."; exit 1; }

echo "Conda environment setup complete."
```
