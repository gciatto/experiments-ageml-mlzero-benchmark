```bash
#!/bin/bash
set -euo pipefail # Exit immediately if a command exits with a non-zero status.
                  # Exit if an unset variable is used.
                  # The return value of a pipeline is the value of the last command to exit with a non-zero status.

# Define paths and versions
CONDA_ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/madelon/rep-1/b94fca9f26a3228d15f1/attempt-1/mlzero/node_7/conda_env"
PYTHON_VERSION="3.11"
COMMON_REQ_PATH="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
ML_REQ_PATH="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/machine learning/requirements.txt"

echo "Starting minimal BASH INSTALLATION script for AutoGluon factory module..."
echo "Target Conda environment path: $CONDA_ENV_PATH"
echo "Requested Python version: $PYTHON_VERSION"

# --- Conda Environment Setup ---

# 1. Ensure conda is initialized for the current shell session
# This is crucial for `conda activate` to work in non-interactive scripts.
# Find the base conda installation path.
CONDA_EXE=$(which conda || true)
if [ -z "$CONDA_EXE" ]; then
    echo "Error: 'conda' command not found in PATH. Please ensure Anaconda/Miniconda is installed and in PATH."
    exit 1
fi

CONDA_BASE=$(dirname $(dirname "$CONDA_EXE"))

# Source the conda.sh script to enable conda commands in the current shell.
if [ -f "$CONDA_BASE/etc/profile.d/conda.sh" ]; then
    echo "Sourcing conda.sh from $CONDA_BASE/etc/profile.d/conda.sh"
    source "$CONDA_BASE/etc/profile.d/conda.sh"
else
    echo "Error: conda.sh not found at $CONDA_BASE/etc/profile.d/conda.sh. Conda initialization failed."
    exit 1
fi

# 2. Create or recreate the conda environment
echo "Attempting to create/recreate conda environment at $CONDA_ENV_PATH..."
# Remove existing environment if it exists to ensure a clean install.
if [ -d "$CONDA_ENV_PATH" ]; then
    echo "Existing environment found at $CONDA_ENV_PATH. Removing it..."
    # Use -y for non-interactive removal.
    # Using 'conda env remove -p' is safer than 'rm -rf' as it handles conda's internal tracking.
    conda env remove -p "$CONDA_ENV_PATH" -y || { echo "Warning: Failed to remove existing environment. Attempting to proceed with creation."; }
fi

# Create the new environment with the specified Python version and pip.
conda create -p "$CONDA_ENV_PATH" python="$PYTHON_VERSION" pip -y
if [ $? -ne 0 ]; then
    echo "Error: Conda environment creation failed."
    exit 1
fi
echo "Conda environment created successfully."

# 3. Activate the environment
echo "Activating conda environment: $CONDA_ENV_PATH"
conda activate "$CONDA_ENV_PATH"
if [ $? -ne 0 ]; then
    echo "Error: Failed to activate conda environment. Check if the environment was created correctly."
    exit 1
fi
echo "Conda environment activated. Python version: $(python --version)"
echo "Pip version in activated environment: $(pip --version)"

# --- Package Installation ---

# 4. Install uv
echo "Installing uv into the activated environment..."
# Use the pip from the activated environment.
pip install uv
echo "uv installed."

# 5. Install required packages from requirements files using uv
echo "Installing packages from requirements files using uv pip install..."
if [ ! -f "$ML_REQ_PATH" ]; then
    echo "Error: ML requirements file not found at $ML_REQ_PATH"
    exit 1
fi
if [ ! -f "$COMMON_REQ_PATH" ]; then
    echo "Error: Common requirements file not found at $COMMON_REQ_PATH"
    exit 1
fi

# Install packages from both requirements files.
# The --prerelease=allow flag is applied to the ML_REQ_PATH as specified.
uv pip install -r "$ML_REQ_PATH" --prerelease=allow -r "$COMMON_REQ_PATH"
if [ $? -ne 0 ]; then
    echo "Error: Failed to install packages from requirements files."
    exit 1
fi
echo "Packages from requirements files installed successfully."

# 6. Install additional packages: numpy, pandas, scikit-learn, and autogluon.tabular
# These are explicitly mentioned in the prompt or inferred from the factory module imports.
echo "Installing numpy, pandas, scikit-learn, and autogluon.tabular..."
uv pip install numpy pandas scikit-learn autogluon.tabular
if [ $? -ne 0 ]; then
    echo "Error: Failed to install core ML libraries (numpy, pandas, scikit-learn, autogluon.tabular)."
    exit 1
fi
echo "Additional core ML packages installed successfully."

echo "All required packages installed. Installation script completed successfully."

# Deactivate the environment to leave the shell in its original state (optional, but good practice).
# Commented out as the execution environment might expect the environment to remain active.
# conda deactivate || true
```
