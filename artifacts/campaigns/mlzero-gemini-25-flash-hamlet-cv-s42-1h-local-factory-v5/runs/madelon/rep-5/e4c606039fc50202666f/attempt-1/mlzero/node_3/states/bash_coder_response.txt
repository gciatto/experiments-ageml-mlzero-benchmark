```bash
#!/bin/bash
set -euo pipefail

# Define paths
# The exact candidate environment path provided in the prompt.
ENV_ROOT="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/madelon/rep-5/e4c606039fc50202666f/attempt-1/mlzero/node_3"
CONDA_ENV_PATH="${ENV_ROOT}/conda_env"
COMMON_REQ="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
TABULAR_REQ="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"

echo "Starting minimal BASH INSTALLATION script for factory module."

# Ensure parent directories exist for the conda environment
mkdir -p "$(dirname "$CONDA_ENV_PATH")"

# 1. Create conda environment with Python 3.11
echo "Creating conda environment at $CONDA_ENV_PATH with Python 3.11..."
conda create -y --prefix "$CONDA_ENV_PATH" python=3.11

# Define paths to the new environment's binaries
PYTHON_BIN="$CONDA_ENV_PATH/bin/python"
PIP_BIN="$CONDA_ENV_PATH/bin/pip"
UV_BIN="$CONDA_ENV_PATH/bin/uv" # uv will be installed here

# Verify Python executable exists
if [ ! -f "$PYTHON_BIN" ]; then
    echo "Error: Python executable not found at $PYTHON_BIN after conda environment creation. Conda environment creation might have failed."
    exit 1
fi

# 2. Install uv into the environment using its pip
echo "Installing uv into the environment using $PIP_BIN..."
"$PIP_BIN" install uv

# Verify uv executable exists
if [ ! -f "$UV_BIN" ]; then
    echo "Error: uv executable not found at $UV_BIN after installation. uv installation might have failed."
    exit 1
fi

# 3. Install required packages using uv
# This command explicitly installs numpy, pandas, scikit-learn as requested,
# and then installs packages from the specified requirements files.
# uv efficiently handles dependency resolution, avoids unnecessary upgrades,
# and respects versions specified in requirements files.
echo "Installing required packages (numpy, pandas, scikit-learn, and from requirements files) using $UV_BIN..."
"$UV_BIN" pip install \
    numpy pandas scikit-learn \
    -r "$TABULAR_REQ" --prerelease=allow \
    -r "$COMMON_REQ"

echo "Conda environment setup complete at $CONDA_ENV_PATH."
echo "The environment is prepared with autogluon.tabular, numpy, pandas, scikit-learn, and other dependencies."
```
