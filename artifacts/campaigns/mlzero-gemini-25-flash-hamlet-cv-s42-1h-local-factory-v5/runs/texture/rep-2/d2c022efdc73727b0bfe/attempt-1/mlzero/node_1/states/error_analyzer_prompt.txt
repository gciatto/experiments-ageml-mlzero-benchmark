        """

        Args:
            time_limit (int): Maximum time in seconds for the AutoGluon fit process.
                              This helps ensure all ten folds finish within the shared deadline.
            random_state (int): Seed for reproducibility of AutoGluon's internal processes.
        """
        self.time_limit = time_limit
        self.random_state = random_state
        self.predictor: TabularPredictor = None
        self.model_path: str = None

    def fit(self, X_train: pd.DataFrame, y_train: pd.Series):
        """
        Fits the AutoGluon TabularPredictor model.

        All preprocessing and model training are performed within this method,
        using only the provided training data. A fresh temporary directory
        is created for AutoGluon artifacts for each fit call, ensuring
        "fresh artifact directories per fold".

        Args:
            X_train (pd.DataFrame): Training feature data.
            y_train (pd.Series): Training target data.
        """
        # Create a unique temporary directory for AutoGluon artifacts for this fold.
        # This ensures "fresh artifact directories per fold" as required.
        self.model_path = tempfile.mkdtemp(prefix="autogluon_model_")

        # AutoGluon expects the target column to be part of the DataFrame.
        train_data = X_train.copy()
        train_data[y_train.name] = y_train

        # Determine problem type (binary or multiclass classification).
        # The task is classification, so we check the number of unique classes.
        problem_type = 'binary' if y_train.nunique() == 2 else 'multiclass'

        # Initialize TabularPredictor with the specified path and random state.
        self.predictor = TabularPredictor(
            label=y_train.name,
            path=self.model_path,
            problem_type=problem_type,
            random_state=self.random_state  # Pass seed for reproducibility
        )

        # Fit the predictor with a time limit.
        # Using 'medium_quality' preset to balance performance and speed,
        # crucial for fitting within the shared deadline.
        # The 'seed' argument is explicitly removed from fit() as per previous error.
        self.predictor.fit(

            time_limit=self.time_limit,
            presets='medium_quality',
        )

    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions using the fitted AutoGluon model.

        Args:
