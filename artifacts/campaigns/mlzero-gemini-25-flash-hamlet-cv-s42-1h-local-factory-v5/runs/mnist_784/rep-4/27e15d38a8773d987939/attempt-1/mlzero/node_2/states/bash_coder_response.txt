```bash
#!/bin/bash

# Exit immediately if a command exits with a non-zero status.
set -e

# Define environment and file paths
CONDA_ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/mnist_784/rep-4/27e15d38a8773d987939/attempt-1/mlzero/node_2/conda_env"
PYTHON_VERSION="3.11"
COMMON_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
MULTIMODAL_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.multimodal/requirements.txt"

echo "Starting minimal BASH INSTALLATION script..."
echo "Conda environment path: $CONDA_ENV_PATH"
echo "Python version: $PYTHON_VERSION"

# 1. Create Conda environment with specified Python version and core libraries
#    numpy, pandas, scikit-learn are explicitly requested to be prepared in the conda_env.
echo "Creating conda environment '$CONDA_ENV_PATH' with Python $PYTHON_VERSION, numpy, pandas, and scikit-learn..."
conda create --prefix "$CONDA_ENV_PATH" python="$PYTHON_VERSION" numpy pandas scikit-learn --yes \
    || { echo "ERROR: Failed to create conda environment." >&2; exit 1; }
echo "Conda environment created successfully."

# 2. Activate the environment
#    Source conda.sh to ensure 'conda activate' is available in non-login shells.
echo "Activating conda environment..."
source "$(conda info --base)/etc/profile.d/conda.sh" \
    || { echo "ERROR: Failed to source conda.sh. Ensure conda is initialized correctly." >&2; exit 1; }
conda activate "$CONDA_ENV_PATH" \
    || { echo "ERROR: Failed to activate conda environment." >&2; exit 1; }
echo "Conda environment activated."

# 3. Install uv (a fast Python package installer)
echo "Installing uv..."
pip install uv \
    || { echo "ERROR: Failed to install uv." >&2; exit 1; }
echo "uv installed successfully."

# 4. Install required packages using uv
#    -r: Install from requirements file.
#    --prerelease=allow: Allow installation of pre-release versions if specified in requirements.
#    The order of requirements files is as specified in the prompt.
echo "Installing packages from requirements files using uv..."
uv pip install -r "$MULTIMODAL_REQUIREMENTS_FILE" --prerelease=allow -r "$COMMON_REQUIREMENTS_FILE" \
    || { echo "ERROR: Failed to install packages from requirements files using uv." >&2; exit 1; }
echo "All required packages installed successfully."

echo "BASH INSTALLATION script completed."
```
