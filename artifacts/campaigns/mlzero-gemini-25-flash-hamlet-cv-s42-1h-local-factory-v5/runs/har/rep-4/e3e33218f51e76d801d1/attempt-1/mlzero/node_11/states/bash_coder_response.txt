```bash
#!/bin/bash
set -euo pipefail

# Define environment path and requirements file paths
CONDA_ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/har/rep-4/e3e33218f51e76d801d1/attempt-1/mlzero/node_11/conda_env"
REQUIREMENTS_COMMON_PATH="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
REQUIREMENTS_ML_PATH="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/machine learning/requirements.txt"

echo "--- Starting Conda environment setup ---"

# 1. Ensure the parent directory for the conda environment exists
mkdir -p "$(dirname "$CONDA_ENV_PATH")"
echo "Created parent directory for conda environment: $(dirname "$CONDA_ENV_PATH")"

# 2. Create the conda environment with Python 3.11 and pip
echo "Creating conda environment at $CONDA_ENV_PATH with Python 3.11..."
conda create -p "$CONDA_ENV_PATH" python=3.11 pip -y || { echo "Conda environment creation failed."; exit 1; }

# 3. Install core scientific packages (numpy, pandas, scikit-learn) using conda for robustness
# This addresses previous ModuleNotFoundError issues for numpy.
echo "Installing numpy, pandas, and scikit-learn using conda..."
conda run -p "$CONDA_ENV_PATH" conda install numpy pandas scikit-learn -y || { echo "Conda install of core packages failed."; exit 1; }

# 4. Install uv using pip within the environment
echo "Installing uv into the environment using pip..."
conda run -p "$CONDA_ENV_PATH" pip install uv || { echo "Installation of uv failed."; exit 1; }

# 5. Install autogluon.tabular and other required packages from requirements.txt files using uv pip install
# This uses 'conda run -p' to ensure commands execute within the target environment,
# addressing previous activation script issues and ModuleNotFoundError for autogluon.
echo "Installing autogluon.tabular and packages from requirements files using uv pip install..."
conda run -p "$CONDA_ENV_PATH" uv pip install \
    autogluon.tabular \
    -r "$REQUIREMENTS_ML_PATH" --prerelease=allow \
    -r "$REQUIREMENTS_COMMON_PATH" || { echo "Installation of autogluon and requirements failed."; exit 1; }

echo "--- Conda environment setup complete ---"
```
