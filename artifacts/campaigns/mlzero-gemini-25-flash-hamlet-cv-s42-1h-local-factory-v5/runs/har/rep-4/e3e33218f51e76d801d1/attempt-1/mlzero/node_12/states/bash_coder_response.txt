```bash
#!/bin/bash
set -euo pipefail

# Define the base directory for the environment
CONDA_ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/har/rep-4/e3e33218f51e76d801d1/attempt-1/mlzero/node_12/conda_env"

# Define paths to requirement files
COMMON_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
AUTOGLUON_TABULAR_REQ_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"

echo "--- Starting Conda Environment Setup ---"
echo "Conda environment path: ${CONDA_ENV_PATH}"

# Ensure conda is available in the PATH
if ! command -v conda &> /dev/null; then
    echo "Error: 'conda' command not found. Please ensure Anaconda/Miniconda is installed and in your PATH."
    exit 1
fi

# Remove existing environment to ensure a clean and consistent installation
if [ -d "${CONDA_ENV_PATH}" ]; then
    echo "Removing existing conda environment at ${CONDA_ENV_PATH}..."
    # Attempt to remove as a conda environment, then fallback to directory removal
    conda env remove --prefix "${CONDA_ENV_PATH}" --yes || {
        echo "Warning: Failed to remove existing conda environment via 'conda env remove'. Attempting directory removal."
        rm -rf "${CONDA_ENV_PATH}" || { echo "Error: Failed to remove directory ${CONDA_ENV_PATH}."; exit 1; }
    }
fi

# Create the conda environment with Python 3.11
echo "Creating conda environment with Python 3.11 at ${CONDA_ENV_PATH}..."
conda create --prefix "${CONDA_ENV_PATH}" python=3.11 --yes || { echo "Error: Conda environment creation failed."; exit 1; }

# Install core packages (numpy, pandas, scikit-learn) using conda
# This addresses previous errors related to numpy and ensures these fundamental libraries are robustly installed.
echo "Installing numpy, pandas, scikit-learn via conda..."
conda run -p "${CONDA_ENV_PATH}" conda install -y numpy pandas scikit-learn || { echo "Error: Failed to install core packages with conda."; exit 1; }

# Install uv into the environment using its python interpreter
# 'conda run -p' ensures the command executes within the specified environment.
echo "Installing uv into the environment..."
conda run -p "${CONDA_ENV_PATH}" python -m pip install uv || { echo "Error: Failed to install uv."; exit 1; }

# Install required packages using uv from the environment
# This includes packages from the specified requirements files.
# 'conda run -p' ensures uv operates within the correct environment.
echo "Installing required packages using uv pip install..."
conda run -p "${CONDA_ENV_PATH}" uv pip install \
    -r "${AUTOGLUON_TABULAR_REQ_FILE}" \
    --prerelease=allow \
    -r "${COMMON_REQ_FILE}" || { echo "Error: Failed to install required packages with uv."; exit 1; }

echo "--- Conda Environment Setup Complete ---"
```
