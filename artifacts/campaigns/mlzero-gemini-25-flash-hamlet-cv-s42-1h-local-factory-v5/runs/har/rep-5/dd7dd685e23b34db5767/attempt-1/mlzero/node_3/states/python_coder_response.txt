```python
import os
import numpy as np
import pandas as pd
import tempfile
import shutil
from autogluon.multimodal import MultiModalPredictor
import warnings

# Suppress warnings from AutoGluon and other libraries for cleaner output.
warnings.filterwarnings('ignore')

# Set a global seed for reproducibility across the module.
SEED = 42
np.random.seed(SEED)

class AutoGluonMultiModalWrapper:
    """
    A wrapper class for AutoGluon MultiModalPredictor to conform to the
    fit/predict interface required by the benchmark harness.
    """
    def __init__(self, label_column_name: str, time_limit: int):
        self.label_column_name = label_column_name
        self.time_limit = time_limit
        self.predictor = None
        self.model_path = None

        # Create a temporary directory for AutoGluon's model artifacts.
        # This ensures a fresh, isolated artifact directory for each fold's model,
        # preventing interference between different training runs.
        self.model_path = tempfile.mkdtemp()
        
        # Initialize MultiModalPredictor. It is unfitted at this stage.
        # AutoGluon is designed to automatically infer data modalities and problem types.
        # Explicitly setting problem_type to "classification" is a good practice
        # to ensure correct behavior for this specific task.
        # No explicit 'presets' are set here, relying on AutoGluon's defaults
        # which are often optimized for general performance and align with guidance.
        self.predictor = MultiModalPredictor(
            label=self.label_column_name,
            path=self.model_path,
            problem_type="classification",
        )

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon MultiModalPredictor using the provided training data.
        All learned preprocessing and model training occur within this method,
        using only the data from the current cross-validation fold.

        Args:
            X (pd.DataFrame): Training features for the current fold.
            y (pd.DataFrame or pd.Series): Training target labels for the current fold.
        """
        # Ensure y is a Series for consistent handling, if it comes as DataFrame
        if isinstance(y, pd.DataFrame):
            # Assuming the target column is the only column in y if it's a DataFrame
            y = y.iloc[:, 0]

        # Combine features and target into a single DataFrame as required by AutoGluon.
        train_data = X.copy()
        train_data[self.label_column_name] = y

        # Fit the predictor with the specified time limit and seed.
        # The `seed` argument ensures reproducibility for AutoGluon's internal processes.
        self.predictor.fit(
            train_data=train_data,
            time_limit=self.time_limit,
            seed=SEED,
            # AutoGluon's presets are already configured according to library guidance.
            # No explicit preset is set here to avoid overriding these defaults,
            # which are often optimized for general performance.
        )

    def predict(self, X: pd.DataFrame) -> pd.Series:
        """
        Predicts class labels for the given input features.

        Args:
            X (pd.DataFrame): Features for which to make predictions.

        Returns:
            pd.Series: Predicted class labels.
        """
        # Make predictions on the input features.
        # AutoGluon handles any necessary preprocessing on X based on its fitted state.
        predictions = self.predictor.predict(X)
        return predictions

    def __del__(self):
        """
        Cleans up the temporary directory created for model artifacts.
        This method is called when the object is garbage collected.
        While `__del__` is not guaranteed to be called in all circumstances,
        it's a common pattern for resource cleanup in Python objects.
        """
        if self.model_path and os.path.exists(self.model_path):
            try:
                shutil.rmtree(self.model_path)
            except OSError as e:
                # Print a warning if cleanup fails, but do not raise an error
                # to avoid disrupting the benchmark process.
                print(f"Warning: Could not remove temporary directory {self.model_path}: {e}")

def build_model(X_train: pd.DataFrame, y_train: pd.Series):
    """
    Generates a fresh, unfitted AutoGluon MultiModal model wrapper.

    This function adheres to the factory contract, returning a new model
    instance with `fit` and `predict` methods for each cross-validation fold.
    The `X_train` and `y_train` arguments are provided by the harness,
    but are primarily used here to infer metadata (like the target column name),
    though 'Class' is hardcoded based on the problem description.

    Args:
        X_train (pd.DataFrame): Training features (used for context, not fitting).
        y_train (pd.Series): Training target (used for context, not fitting).

    Returns:
        AutoGluonMultiModalWrapper: An unfitted model wrapper ready for training.
    """
    # The target column name is explicitly stated as 'Class' in the README.md.
    label_column_name = "Class"
    
    # Calculate a time limit per fold to ensure all ten fits complete within
    # the total shared deadline. The total deadline is 1067 seconds for 10 folds.
    # Allocating 60 seconds per fold (60 * 10 = 600 seconds total for fitting)
    # leaves ample time (467 seconds) for overhead, data loading, and other operations.
    time_limit_per_fold = 60 

    return AutoGluonMultiModalWrapper(
        label_column_name=label_column_name,
        time_limit=time_limit_per_fold
    )
```
