```python
import pandas as pd
from autogluon.tabular import TabularPredictor
import tempfile
import shutil
import logging
import os
import random
import numpy as np

# --- Module-level Constants and Configuration ---
# Seed for reproducibility. Used for global random number generators.
RANDOM_SEED = 42
random.seed(RANDOM_SEED)  # Set Python's built-in random seed
np.random.seed(RANDOM_SEED)  # Set NumPy's random seed

# The name of the target column as specified in the problem description.
TARGET_COLUMN = 'Class'

# Time limit for AutoGluon's fit method in seconds.
# The total shared deadline is 2857 seconds for 10 folds.
# Allocating 280 seconds per fold (2857 / 10 = 285.7) leaves a buffer for overhead.
FIT_TIME_LIMIT_SECONDS = 280

# Configure logging to suppress excessive AutoGluon output if desired.
# These lines are commented out by default to avoid interfering with the harness's
# logging configuration, but can be uncommented for debugging.
# logging.basicConfig(level=logging.INFO)
# logging.getLogger('autogluon').setLevel(logging.WARNING)

# --- Model Wrapper Class ---
class AutoGluonModelWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor, conforming to the
    fit/predict interface required by the benchmark harness.
    """
    def __init__(self):
        """
        Initializes the wrapper. The AutoGluon predictor itself is initialized
        and fitted only when the `fit` method is called.
        """
        self.predictor = None
        self.model_path = None  # Path to store AutoGluon's artifacts for this model instance

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon TabularPredictor using the provided training data.
        All preprocessing and model training are handled by AutoGluon internally.
        A unique temporary directory is created for AutoGluon artifacts for each fit call.

        Args:
            X (pd.DataFrame): Training features.
            y (pd.Series): Training target labels.
        """
        # Ensure y is a Series with the correct name for AutoGluon
        if not isinstance(y, pd.Series):
            y = pd.Series(y, name=TARGET_COLUMN, index=X.index)
        else:
            y.name = TARGET_COLUMN

        # Combine X and y into a single DataFrame, which AutoGluon expects
        train_data = pd.concat([X, y], axis=1)

        # Create a temporary directory for AutoGluon artifacts.
        # This ensures "fresh artifact directories per fold" as required.
        # The directory will be cleaned up on object deletion (best effort) or by the OS.
        self.model_path = tempfile.mkdtemp(prefix="autogluon_model_")
        # logging.info(f"AutoGluon model artifacts path: {self.model_path}")

        # Initialize TabularPredictor.
        # The 'random_state' argument is removed from the constructor as it's not
        # accepted by the installed AutoGluon version (as per previous errors).
        # Global seeds for random and numpy are set at the module level for reproducibility.
        # eval_metric is set to 'balanced_accuracy' as per task requirements.
        self.predictor = TabularPredictor(
            label=TARGET_COLUMN,
            path=self.model_path,
            eval_metric='balanced_accuracy',
            # random_state=RANDOM_SEED # Removed as per previous error fix
        )

        # Fit the predictor.
        # time_limit ensures that each fit operation respects the overall deadline.
        # presets='best_quality' aims for the best performance given the time limit.
        self.predictor.fit(
            train_data=train_data,
            time_limit=FIT_TIME_LIMIT_SECONDS,
            presets='best_quality',
            # The 'seed' argument is not accepted by TabularPredictor.fit(),
            # as noted in previous errors. Randomness is controlled via global seeds
            # and potentially internal AutoGluon mechanisms.
        )

    def predict(self, X: pd.DataFrame) -> pd.Series:
        """
        Makes predictions on new data using the fitted AutoGluon model.

        Args:
            X (pd.DataFrame): Features for which to make predictions.

        Returns:
            pd.Series: Predicted class labels.
        """
        if self.predictor is None:
            raise RuntimeError("Model has not been fitted yet. Call fit(X, y) first.")
        return self.predictor.predict(X)

    def __del__(self):
        """
        Attempts to clean up the temporary directory created for AutoGluon artifacts
        when the model wrapper object is garbage collected. This is a best-effort
        cleanup and not guaranteed to run in all circumstances.
        """
        if self.model_path and os.path.exists(self.model_path):
            try:
                shutil.rmtree(self.model_path)
                # logging.info(f"Cleaned up AutoGluon model path: {self.model_path}")
            except OSError as e:
                # logging.warning(f"Error cleaning up AutoGluon model path {self.model_path}: {e}")
                pass

# --- Factory Function ---
def build_model(X_train: pd.DataFrame, y_train: pd.Series) -> AutoGluonModelWrapper:
    """
    Factory function to generate a new, unfitted model wrapper instance.
    This function adheres to the contract required by the benchmark harness.

    Args:
        X_train (pd.DataFrame): Training features (provided for context, not used for fitting here).
        y_train (pd.Series): Training target labels (provided for context, not used for fitting here).

    Returns:
        AutoGluonModelWrapper: An unfitted instance of the model wrapper.
    """
    # X_train and y_train are provided for context but should not be used
    # to fit anything at this stage, as per the "unfitted" requirement.
    return AutoGluonModelWrapper()
```
