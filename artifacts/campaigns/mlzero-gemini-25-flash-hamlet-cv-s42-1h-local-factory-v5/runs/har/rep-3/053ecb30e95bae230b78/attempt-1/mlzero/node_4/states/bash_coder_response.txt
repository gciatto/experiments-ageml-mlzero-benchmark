```bash
#!/bin/bash
set -euo pipefail

# Define the target conda environment directory
CONDA_ENV_DIR="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/har/rep-3/053ecb30e95bae230b78/attempt-1/mlzero/node_4/conda_env"

# Define paths to the requirements files
COMMON_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
ML_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/machine learning/requirements.txt"

echo "--- Starting minimal BASH INSTALLATION script ---"

# 1. Ensure a clean slate: Remove any existing environment at the target path
echo "Removing existing conda environment (if any) at '$CONDA_ENV_DIR'..."
conda env remove -p "$CONDA_ENV_DIR" -y || true

# 2. Create the conda environment with Python 3.11
echo "Creating conda environment at '$CONDA_ENV_DIR' with Python 3.11..."
conda create -p "$CONDA_ENV_DIR" python=3.11 -y

# 3. Verify environment creation and activate it
echo "Activating conda environment..."
# Source conda.sh to make 'conda' command available in non-interactive shell
if [ -f "$(conda info --base)/etc/profile.d/conda.sh" ]; then
    source "$(conda info --base)/etc/profile.d/conda.sh"
else
    echo "ERROR: conda.sh not found. Ensure Conda is installed and initialized in your shell." >&2
    exit 1
fi

# Check if the activate script exists after creation, as per previous error fix
if [ ! -f "$CONDA_ENV_DIR/bin/activate" ]; then
    echo "ERROR: Conda environment activation script not found at '$CONDA_ENV_DIR/bin/activate'. Conda environment creation might have failed." >&2
    exit 1
fi

conda activate "$CONDA_ENV_DIR"

# 4. Install uv using the environment's pip
echo "Installing uv into the activated environment..."
pip install uv

# 5. Install required packages from specified requirements.txt files using uv
echo "Installing packages from requirements files using uv pip install..."
uv pip install -r "$ML_REQUIREMENTS_FILE" --prerelease=allow -r "$COMMON_REQUIREMENTS_FILE"

# 6. Install additional packages: numpy, pandas, scikit-learn, and autogluon.tabular
# These are explicitly requested and/or derived from the factory module's imports.
echo "Installing numpy, pandas, scikit-learn, and autogluon.tabular using uv pip install..."
uv pip install numpy pandas scikit-learn autogluon.tabular

echo "--- Conda environment setup complete for '$CONDA_ENV_DIR' ---"
```
