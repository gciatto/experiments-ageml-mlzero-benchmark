class AutoGluonModelWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor, conforming to the
    fit/predict interface required by the benchmark harness.
    """
    def __init__(self):
        """
        Initializes the wrapper. The AutoGluon predictor itself is initialized
        and fitted only when the `fit` method is called.
        """
        self.predictor = None
        self.model_path = None  # Path to store AutoGluon's artifacts for this model instance

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon TabularPredictor using the provided training data.
        All preprocessing and model training are handled by AutoGluon internally.
        A unique temporary directory is created for AutoGluon artifacts for each fit call.

        Args:
            X (pd.DataFrame): Training features.
            y (pd.Series): Training target labels.
        """
        # Ensure y is a Series with the correct name for AutoGluon
        if not isinstance(y, pd.Series):
            y = pd.Series(y, name=TARGET_COLUMN, index=X.index)
        else:
            y.name = TARGET_COLUMN

        # Combine X and y into a single DataFrame, which AutoGluon expects
        train_data = pd.concat([X, y], axis=1)

        # Create a temporary directory for AutoGluon artifacts.
        # This ensures "fresh artifact directories per fold" as required.
        # The directory will be cleaned up on object deletion (best effort) or by the OS.
        self.model_path = tempfile.mkdtemp(prefix="autogluon_model_")

        # Initialize TabularPredictor.
        # The 'random_state' argument is removed from the constructor as it's not
        # accepted by the installed AutoGluon version (as per previous errors).

        # eval_metric is set to 'balanced_accuracy' as per task requirements.
        self.predictor = TabularPredictor(
            label=TARGET_COLUMN,

            # as noted in previous errors. Randomness is controlled via global seeds
            # and potentially internal AutoGluon mechanisms.
        )
