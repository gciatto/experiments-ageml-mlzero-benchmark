```bash
#!/bin/bash
set -e

# Define the full path for the conda environment as specified for node_3
CONDA_ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/har/rep-3/053ecb30e95bae230b78/attempt-1/mlzero/node_3/conda_env"

# Define paths to the requirements files
COMMON_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
AUTOGLUON_TABULAR_REQUIREMENTS_FILE="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"

echo "--- Starting Conda environment setup for node_3 ---"

# Ensure conda is initialized for the current shell session
# This is crucial for 'conda activate' to work in non-interactive scripts.
source "$(conda info --base)/etc/profile.d/conda.sh"

# 1. Remove any existing environment at the target path to ensure a clean slate.
# This addresses potential issues from previous failed attempts and ensures a fresh environment.
if [ -d "$CONDA_ENV_PATH" ]; then
    echo "Removing existing conda environment at '$CONDA_ENV_PATH'..."
    conda env remove --prefix "$CONDA_ENV_PATH" --yes || true # '|| true' prevents script exit if env doesn't exist or removal fails non-critically
fi

# 2. Create a new conda environment with Python 3.11
echo "Creating conda environment at '$CONDA_ENV_PATH' with Python 3.11..."
conda create --prefix "$CONDA_ENV_PATH" python=3.11 --yes

# 3. Activate the newly created environment
echo "Activating conda environment..."
conda activate "$CONDA_ENV_PATH"

# Verify that the environment was successfully activated by checking the Python executable path.
if [[ "$(which python)" != "$CONDA_ENV_PATH/bin/python" ]]; then
    echo "ERROR: Conda environment activation failed. Python executable not found in '$CONDA_ENV_PATH/bin/'."
    exit 1
fi

# 4. Install uv (a fast Python package installer) into the activated environment
echo "Installing uv into the environment..."
pip install uv

# 5. Install required packages using uv
# The command installs packages from both requirements files, allowing prereleases.
# uv ensures only specified packages and their dependencies are installed,
# and will not upgrade/reinstall if already at the correct version.
# The order of requirements files is specified in the prompt: autogluon.tabular first, then _common.
echo "Installing packages from requirements files using uv pip install..."
uv pip install \
    -r "$AUTOGLUON_TABULAR_REQUIREMENTS_FILE" \
    --prerelease=allow \
    -r "$COMMON_REQUIREMENTS_FILE"

echo "--- Conda environment setup complete for node_3 ---"
```
