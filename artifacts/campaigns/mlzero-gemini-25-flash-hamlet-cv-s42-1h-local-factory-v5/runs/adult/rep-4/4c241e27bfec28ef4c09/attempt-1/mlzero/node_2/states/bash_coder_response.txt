```bash
#!/bin/bash
set -euo pipefail

# Define the target conda environment path
CONDA_ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/adult/rep-4/4c241e27bfec28ef4c09/attempt-1/mlzero/node_2/conda_env"

# Define paths to requirement files
REQUIREMENTS_COMMON="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
REQUIREMENTS_MULTIMODAL="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.multimodal/requirements.txt"

echo "--- Starting Conda Environment Setup ---"

# 1. Create a new conda environment with Python 3.11
echo "Creating conda environment at $CONDA_ENV_PATH with Python 3.11..."
conda create -p "$CONDA_ENV_PATH" python=3.11 -y

# 2. Activate the environment
echo "Activating conda environment..."
source "$(conda info --base)/etc/profile.d/conda.sh"
conda activate "$CONDA_ENV_PATH"

# 3. Install uv
echo "Installing uv into the environment..."
pip install uv

# 4. Install required packages from requirements.txt files using uv
echo "Installing packages from requirements files using uv..."
uv pip install -r "$REQUIREMENTS_MULTIMODAL" --prerelease=allow -r "$REQUIREMENTS_COMMON"

# 5. Install numpy, pandas, and scikit-learn using uv
echo "Installing numpy, pandas, and scikit-learn using uv..."
uv pip install numpy pandas scikit-learn

echo "--- Conda Environment Setup Complete ---"
```
