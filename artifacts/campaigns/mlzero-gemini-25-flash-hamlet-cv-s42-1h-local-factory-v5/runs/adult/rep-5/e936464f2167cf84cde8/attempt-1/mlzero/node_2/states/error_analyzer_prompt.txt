class AutoGluonTabularWrapper:

    A wrapper class for AutoGluon's TabularPredictor to conform to the

        self.random_seed = random_seed
        self.predictor = None
        self.model_path = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon TabularPredictor model.

        All preprocessing, feature engineering, and model training are
        handled by AutoGluon internally and are fitted exclusively on
        the provided training data (X, y) for the current fold.

        Args:

            y (pd.Series): Training target labels.
        """
        # Combine features and target into a single DataFrame as required by AutoGluon.

        train_data[self.label_column] = y
