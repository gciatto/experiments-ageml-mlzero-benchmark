import pandas as pd
import numpy as np
import random
import os
import uuid
import shutil
from pathlib import Path
from autogluon.tabular import TabularPredictor

# Set global seeds for reproducibility across the module.
# AutoGluon's TabularPredictor also takes a random_seed argument for internal reproducibility.
SEED = 42
np.random.seed(SEED)
random.seed(SEED)

class AutoGluonTabularWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor to conform to the
    fit/predict interface required by the benchmark harness.
    """
    def __init__(self, label_column: str, random_seed: int):
        """
        Initializes the wrapper. The AutoGluon predictor itself is not
        instantiated or fitted until the .fit() method is called.

        Args:
            label_column (str): The name of the target column.
            random_seed (int): The random seed for reproducibility.
        """
        self.label_column = label_column
        self.random_seed = random_seed
        self.predictor = None
        self.model_path = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon TabularPredictor model.

        All preprocessing, feature engineering, and model training are
        handled by AutoGluon internally and are fitted exclusively on
        the provided training data (X, y) for the current fold.

        Args:
            X (pd.DataFrame): Training features.
            y (pd.Series): Training target labels.
        """
        # Combine features and target into a single DataFrame as required by AutoGluon.
        train_data = X.copy()
        train_data[self.label_column] = y

        # Generate a unique path for AutoGluon artifacts for this specific fit call.
        # This ensures "fresh artifact directories per fold" as required.
        # The path is relative to the current working directory, which will be
        # the node output folder where model_factory.py is saved by the harness.
        unique_id = uuid.uuid4().hex
        self.model_path = Path(f'./autogluon_models_{unique_id}')
        
        # Ensure the directory is clean before AutoGluon uses it.
        # This prevents issues if a directory with the same name (highly unlikely with uuid)
        # or previous artifacts somehow existed, ensuring a truly fresh start for each fold.
        if self.model_path.exists():
            shutil.rmtree(self.model_path)
        
        # Initialize TabularPredictor.
        # The 'path' argument directs AutoGluon to store its models and intermediate
        # files in the unique directory created for this fold.
        # 'random_seed' is passed to the constructor for internal reproducibility.
        self.predictor = TabularPredictor(
            label=self.label_column,
            path=str(self.model_path),
            random_seed=self.random_seed
        )

        # Set a time limit for fitting to ensure all ten folds complete within
        # the total shared deadline.
        # Given a total remaining time of 3235 seconds for 10 folds,
        # an average of 323.5 seconds per fold is available.
        # A conservative limit of 250 seconds per fold is chosen to account for
        # overhead (data loading, model instantiation, prediction, evaluation)
        # and potential variance in training times across folds.
        time_limit_per_fold = 250 # seconds

        # Fit the predictor. AutoGluon automatically handles model selection,
        # hyperparameter tuning, and ensemble creation within the given time limit.
        self.predictor.fit(
            train_data=train_data,
            time_limit=time_limit_per_fold,
            # For classification, AutoGluon defaults to 'balanced_accuracy' for evaluation
            # during training if not specified, which aligns with the problem's metric.
            # Consider using presets like 'best_quality' or 'medium_quality_faster_inference_only_refit_full'
            # if more control over model quality vs. speed is needed, but 'time_limit' is primary.
            # presets='medium_quality_faster_inference_only_refit_full',
            # ag_args_fit={'num_gpus': 0} # Uncomment and set to >0 if GPUs are available
        )

    def predict(self, X: pd.DataFrame) -> pd.Series:
        """
        Makes predictions using the fitted AutoGluon TabularPredictor model.

        Args:
            X (pd.DataFrame): Features for which to make predictions.

        Returns:
            pd.Series: Predicted class labels.
        """
        if self.predictor is None:
            raise RuntimeError("Model has not been fitted yet. Call .fit(X, y) first.")
        
        # AutoGluon's predict method expects a DataFrame and returns a Series of predictions.
        predictions = self.predictor.predict(X)
        return predictions

def build_model(X_train: pd.DataFrame, y_train: pd.Series) -> AutoGluonTabularWrapper:
    """
    Factory function to create a new, unfitted AutoGluonTabularWrapper model.

    This function is called by the harness for each fold to obtain a fresh
    model instance. The model is not fitted here; fitting occurs when the
    returned wrapper's .fit(X, y) method is invoked.

    Args:
        X_train (pd.DataFrame): Example training features. Provided as a hint
                                for model structure (e.g., column names, types).
        y_train (pd.Series): Example training target. Provided as a hint for
                             the target column name and type.

    Returns:
        AutoGluonTabularWrapper: A new, unfitted model wrapper instance.
    """
    # The target column name is explicitly 'class' as per the problem description.
    target_column_name = 'class'
    
    # Return a new, unfitted wrapper instance.
    return AutoGluonTabularWrapper(label_column=target_column_name, random_seed=SEED)
