```bash
#!/bin/bash
set -e

# Define the target conda environment path for node_9
ENV_PATH="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/ames_housing/rep-3/5a9401ba768907dcd35e/attempt-1/mlzero/node_9/conda_env"

# Define paths to the requirements files
REQ_COMMON="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"
REQ_AUTOGLUON_TABULAR="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"

echo "Starting minimal BASH INSTALLATION script for AutoGluonTabularWrapper (node_9)..."

# Ensure the parent directory exists before creating the environment
mkdir -p "$(dirname "$ENV_PATH")"

# 1. Create the conda environment with Python 3.11
echo "Creating conda environment at '$ENV_PATH' with Python 3.11..."
conda create -p "$ENV_PATH" python=3.11 -y

# 2. Initialize conda in the current shell for 'conda activate' to work reliably.
# This addresses previous errors related to conda activation issues (e.g., "conda init" or "unbound variable").
# We assume 'conda' is already in PATH and 'conda info --base' works to find the base installation.
echo "Initializing conda shell hooks for the script..."
source "$(conda info --base)/etc/profile.d/conda.sh"

# 3. Activate the newly created environment
echo "Activating conda environment '$ENV_PATH'..."
conda activate "$ENV_PATH"

# 4. Install 'uv' into the active environment
echo "Installing 'uv' into the environment..."
# Now that the environment is active, 'pip' refers to the environment's pip.
pip install uv

# 5. Install required packages using 'uv'
# This command installs packages from both requirements files and explicitly adds numpy, pandas, and scikit-learn.
# 'uv' will resolve all dependencies, ensuring only specified packages and their dependencies are installed.
# The '--prerelease=allow' flag is included as specified.
# Now that 'uv' is installed and the environment is active, we can just call 'uv'.
echo "Installing packages from requirements files and additional libraries (numpy, pandas, scikit-learn) using uv..."
uv pip install \
    -r "$REQ_AUTOGLUON_TABULAR" \
    --prerelease=allow \
    -r "$REQ_COMMON" \
    numpy pandas scikit-learn

echo "Conda environment setup complete at '$ENV_PATH'."
echo "The environment is prepared with its own interpreter/pip for subsequent operations."
```
