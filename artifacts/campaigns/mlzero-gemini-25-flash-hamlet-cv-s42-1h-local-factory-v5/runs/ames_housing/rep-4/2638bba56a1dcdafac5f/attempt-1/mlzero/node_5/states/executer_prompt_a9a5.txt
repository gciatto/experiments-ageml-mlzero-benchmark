import pandas as pd
import logging

from autogluon.tabular import TabularPredictor, TabularDataset

# Set up logging for the module. This will output informational messages
# about the model's operations, such as temporary directory creation.
logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)

# Global seed for reproducibility. This ensures that any random operations
# outside of AutoGluon's explicit control are consistent.

np.random.seed(GLOBAL_SEED)

class AutoGluonTabularWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor to conform to the
    `fit(X, y)` and `predict(X)` interface required by the evaluation harness.
    Each instance of this wrapper manages its own AutoGluon model and
    associated temporary files.
    """

    def __init__(self, time_limit_per_fold: int):
        """
        Initializes the wrapper. The AutoGluon predictor itself is not
        instantiated or fitted until the `fit` method is called.

        Args:
            time_limit_per_fold (int): The maximum time in seconds allowed
                                       for AutoGluon's `fit` operation for
                                       a single fold. This helps ensure all
                                       ten folds complete within the overall
                                       shared deadline.
        """
        self.predictor = None
        self.model_path = None
        self.time_limit_per_fold = time_limit_per_fold
        # The target column name for the Ames Housing dataset.
        self.label_column = 'SalePrice'

    def fit(self, X: pd.DataFrame, y: pd.Series):
        """
        Fits the AutoGluon TabularPredictor model using the provided training data.
        All preprocessing, model selection, and training are handled by AutoGluon
        within this method, using only the data from the current fold.

        Args:
            X (pd.DataFrame): Training features for the current fold.
            y (pd.Series): Training target values for the current fold.
        """
        # Combine X and y into a single DataFrame, as AutoGluon's TabularDataset
        # expects the label column to be present in the input DataFrame.
        train_df = X.copy()
        train_df[self.label_column] = y

        # Create a unique temporary directory for AutoGluon's model artifacts
        # for this specific fold. This ensures "fresh artifact directories per fold"
        # and prevents conflicts between different folds or runs.
        self.model_path = tempfile.mkdtemp(prefix="autogluon_ames_housing_model_")
        logger.info(f"AutoGluon model artifacts for this fold will be saved to: {self.model_path}")

        # Initialize TabularPredictor.
        # - `label`: Specifies the target column.
        # - `path`: The directory where AutoGluon will save its models and logs.
        # - `eval_metric`: Set to 'root_mean_squared_error' as required by the task.
        # - `problem_type`: Explicitly set to 'regression' for clarity, though
        #                   AutoGluon often infers this correctly for numerical labels.
        self.predictor = TabularPredictor(
            label=self.label_column,
            path=self.model_path,
            eval_metric='root_mean_squared_error',
            problem_type='regression',
        )

        # Fit the predictor.
        # - `train_data`: The combined DataFrame containing features and target.
        # - `time_limit`: Enforces a maximum training time per fold, crucial for
        #                 meeting the overall deadline.
        # - `presets`: 'high_quality' offers a good balance between predictive
        #              accuracy and training time, suitable for the given time constraints.
        # - `random_seed`: This argument was removed as per the fix. AutoGluon's
        #                  internal randomness is controlled by AG_RANDOM_SEED env var.
        self.predictor.fit(
            train_data=TabularDataset(train_df),
            time_limit=self.time_limit_per_fold,
            presets='high_quality', # Retaining 'high_quality' for balance between accuracy and speed.
            # AutoGluon automatically handles internal validation splits,
            # so there's no need to manually split X_train, y_train further.
        )
        logger.info(f"AutoGluon model fitted successfully for this fold.")

    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions using the fitted AutoGluon TabularPredictor model.

        Args:
            X (pd.DataFrame): Features for which to make predictions.

        Returns:
            np.ndarray: A 1D numpy array of predicted `SalePrice` values.
        """
        if self.predictor is None:
            raise RuntimeError("Model has not been fitted yet. Call .fit(X, y) first.")

        # AutoGluon's `predict` method expects a TabularDataset or DataFrame.
        predictions = self.predictor.predict(TabularDataset(X))

        # Ensure the output is a 1D numpy array as required by the contract.
        return predictions.to_numpy()

    def __del__(self):
        """
        Attempts to clean up the temporary directory created for AutoGluon artifacts.
        This method is called when the object is garbage collected. While its
        execution is not strictly guaranteed in all Python environments or
        program termination scenarios, it's a best-effort attempt to manage
        disk space.
        """
        if self.model_path and os.path.exists(self.model_path):
            try:
                shutil.rmtree(self.model_path)
                logger.info(f"Cleaned up AutoGluon model artifacts from: {self.model_path}")
            except OSError as e:
                logger.warning(f"Error cleaning up model path {self.model_path}: {e}")
