        logging.info(f"Created temporary model path for AutoGluon MultiModal: {self.model_path}")
        
        # Combine X and y into a single DataFrame, as MultiModalPredictor expects this format.
        train_df = X.copy()
        train_df[self.target_column] = y

        # Drop the 'Id' column if it exists, as it's explicitly stated not to be a feature.
        if 'Id' in train_df.columns:
            train_df = train_df.drop(columns=['Id'])

            label=self.target_column,
            problem_type='regression',
            path=self.model_path,
            seed=self.seed,

        logging.info("AutoGluon MultiModalPredictor fit completed successfully.")
