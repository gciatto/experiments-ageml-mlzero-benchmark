```python
import pandas as pd
import numpy as np
import os
import tempfile
import shutil
import logging
from autogluon.tabular import TabularPredictor

# Configure logging to suppress excessive AutoGluon output if desired.
# For a benchmark, it's often useful to see full logs, but if they are too verbose,
# uncomment and adjust the following line:
# logging.getLogger('autogluon').setLevel(logging.WARNING)

class AutoGluonModelWrapper:
    """
    A wrapper class for AutoGluon's TabularPredictor, providing a scikit-learn-like
    fit/predict interface and handling specific task requirements for the benchmark.
    """
    def __init__(self, path_prefix="autogluon_artifacts", random_state=42, time_limit=15): # Adjusted time_limit for faster execution
        """
        Initializes the wrapper. The AutoGluon predictor itself is initialized
        and fitted within the .fit() method.

        Args:
            path_prefix (str): Prefix for the temporary directory where AutoGluon
                               artifacts will be stored for each fold.
            random_state (int): Seed for reproducibility of stochastic elements
                                within AutoGluon.
            time_limit (int): Maximum time in seconds for AutoGluon's fit process
                              for a single fold. This is crucial for meeting the
                              overall deadline across all ten folds.
        """
        self.predictor = None
        self.path_prefix = path_prefix
        self.random_state = random_state
        self.time_limit = time_limit
        self.model_path = None  # Stores the unique path for this model instance's artifacts

    def fit(self, X_train: pd.DataFrame, y_train: pd.Series):
        """
        Fits the AutoGluon TabularPredictor on the provided training data.
        Handles feature engineering (dropping 'Id') and target transformation (log1p).

        Args:
            X_train (pd.DataFrame): Training features for the current fold.
            y_train (pd.Series): Training target variable for the current fold.
        """
        # Create a unique temporary directory for this fold's artifacts.
        # This ensures "fresh artifact directories per fold" as required by the contract.
        self.model_path = tempfile.mkdtemp(prefix=self.path_prefix)

        # Drop 'Id' column if it exists, as it must not be used as a feature.
        X_train_processed = X_train.drop(columns=['Id'], errors='ignore')

        # Apply log transformation (log1p) to the target variable.
        # SalePrice is typically right-skewed, and log transformation often
        # helps models perform better for RMSE metric.
        y_train_transformed = np.log1p(y_train)

        # Initialize TabularPredictor.
        # 'label' is a placeholder; it will be replaced by the actual column name 'target'
        # when train_data is created.
        # FIX: Removed 'random_state' from TabularPredictor constructor as it's not accepted.
        self.predictor = TabularPredictor(
            label='target',
            path=self.model_path,
            problem_type='regression',
            eval_metric='rmse',  # Explicitly set RMSE as the evaluation metric.
        )

        # Combine X and y into a single DataFrame for AutoGluon.
        train_data = X_train_processed.copy()
        train_data['target'] = y_train_transformed

        # Fit the predictor with the specified time limit and quality preset.
        # 'best_quality' preset aims for the best possible model performance.
        # FIX: Pass 'random_state' via 'ag_args_fit' to the fit method.
        self.predictor.fit(
            train_data=train_data,
            time_limit=self.time_limit,
            presets='best_quality',
            ag_args_fit={'random_state': self.random_state} # Correct way to pass random_state for reproducibility
        )

    def predict(self, X_test: pd.DataFrame) -> np.ndarray:
        """
        Makes predictions on new data using the fitted AutoGluon model.
        Handles feature engineering (dropping 'Id') and inverse target transformation.

        Args:
            X_test (pd.DataFrame): Test features for which to make predictions.

        Returns:
            np.ndarray: Predicted continuous numerical values for 'SalePrice',
                        transformed back to the original scale.
        """
        if self.predictor is None:
            raise RuntimeError("Model has not been fitted yet. Call .fit() first.")

        # Drop 'Id' column if it exists, consistent with training preprocessing.
        X_test_processed = X_test.drop(columns=['Id'], errors='ignore')

        # Predict on the processed test data.
        predictions_transformed = self.predictor.predict(X_test_processed)

        # Apply inverse log transformation (expm1) to return predictions
        # in the original 'SalePrice' scale.
        predictions = np.expm1(predictions_transformed)

        return predictions.values  # Return as a numpy array.

    def __del__(self):
        """
        Attempts to clean up the temporary directory created for AutoGluon artifacts
        when the AutoGluonModelWrapper object is garbage collected.
        Note: The execution of __del__ is not guaranteed by Python, especially
        in cases of abnormal program termination or circular references.
        """
        if self.model_path and os.path.exists(self.model_path):
            try:
                shutil.rmtree(self.model_path)
            except OSError as e:
                # Log or ignore errors during cleanup.
                # print(f"Error cleaning up AutoGluon artifacts at {self.model_path}: {e}")
                pass


def build_model(X_train: pd.DataFrame, y_train: pd.Series) -> AutoGluonModelWrapper:
    """
    Factory function to create a new, unfitted AutoGluon model wrapper.

    This function adheres to the contract: it returns a fresh, unfitted model
    wrapper that exposes `fit(X,y)` and `predict(X)` methods.
    All preprocessing and model fitting are encapsulated within the wrapper's methods,
    ensuring they are fitted only on the training data of each fold.

    Args:
        X_train (pd.DataFrame): Training features (provided for context, but not
                                used for fitting within this function call).
        y_train (pd.Series): Training target (provided for context, but not
                             used for fitting within this function call).

    Returns:
        AutoGluonModelWrapper: An unfitted model wrapper instance ready to be fitted.
    """
    # Instantiate the wrapper. The actual fitting will happen when its .fit() method is called
    # by the evaluation harness. The random_state and time_limit are set here for consistency
    # across all folds.
    # The time_limit is set to 15 seconds per fold to ensure all ten fits finish within
    # the remaining shared deadline (219 seconds total).
    return AutoGluonModelWrapper(random_state=42, time_limit=15)
```
