```bash
#!/bin/bash
set -euo pipefail

# Define environment and requirements paths
ENV_DIR="/root/experiments-ageml-mlzero-benchmark/artifacts/campaigns/mlzero-gemini-25-flash-hamlet-cv-s42-1h-local-factory-v5/runs/ames_housing/rep-1/4d252b8d3aa6da775378/attempt-1/mlzero/node_0/conda_env"
AUTOGLUON_TABULAR_REQUIREMENTS="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/autogluon.tabular/requirements.txt"
COMMON_REQUIREMENTS="/root/experiments-ageml-mlzero-benchmark/.mlzero-local/source/src/autogluon/assistant/tools_registry/_common/requirements.txt"

echo "--- Starting Conda environment setup ---"

# 1. Create Conda environment with Python 3.11
echo "Creating conda environment at '$ENV_DIR' with Python 3.11..."
conda create -p "$ENV_DIR" python=3.11 -y

# 2. Activate the environment
echo "Activating environment '$ENV_DIR'..."
# Source conda.sh to make 'conda activate' available in non-interactive scripts
eval "$(conda shell hook)"
conda activate "$ENV_DIR"

# 3. Install uv
echo "Installing uv into the environment..."
pip install uv

# 4. Install required packages using uv
echo "Installing required packages from '$AUTOGLUON_TABULAR_REQUIREMENTS' and '$COMMON_REQUIREMENTS' using uv..."
uv pip install \
    -r "$AUTOGLUON_TABULAR_REQUIREMENTS" \
    --prerelease=allow \
    -r "$COMMON_REQUIREMENTS"

echo "--- Conda environment setup complete ---"
```
